472	/// Borderless panel with the translucent menu look, rounded corners and shadow.
473	final class MenuPanel: NSPanel {
474	    override var canBecomeKey: Bool { true }   // borderless panels can't otherwise take typing (search)
475	    override func cancelOperation(_ sender: Any?) { MenuController.shared.hide() }   // Esc closes it
476	
477	    /// Keep the top edge pinned when the content resizes (window shade), instead of the bottom.
478	    override func setFrame(_ frameRect: NSRect, display flag: Bool) {
479	        var rect = frameRect
480	        if isVisible, rect.size != frame.size {
481	            rect.origin = NSPoint(x: frame.minX, y: frame.maxY - rect.height)
482	        }
483	        super.setFrame(rect, display: flag)
484	        if isVisible { refreshShadow() }
485	    }

487	    /// A clear borderless window can leave its old shadow on screen after it moves or resizes
488	    /// (a ghost the compositor shows but screenshots don't). Rebuild the shadow from scratch.
489	    func refreshShadow() {
490	        hasShadow = false
491	        hasShadow = true
492	        invalidateShadow()
493	    }

496	        let panel = MenuPanel(contentRect: .zero,
497	                              styleMask: [.borderless, .nonactivatingPanel, .fullSizeContentView],
498	                              backing: .buffered, defer: false)

500	        panel.level = .floating
501	        panel.hidesOnDeactivate = false
502	        panel.isReleasedWhenClosed = false
503	        panel.isOpaque = false

564	/// Channel logos and album art kept in memory once loaded. Nothing is fetched until it's shown,
565	/// and a URL that is already loading is shared instead of downloaded twice. Each image is fetched
566	/// (where the server can resize) and decoded at 2x the points it is shown at, so the 32pt channel
567	/// rows and the 64pt Now Playing tile each get their own size, and the cache is bounded.
568	@MainActor
569	final class ArtCache {
570	    static let shared = ArtCache()
571	    private let images: NSCache<NSString, NSImage> = {
572	        let cache = NSCache<NSString, NSImage>()
573	        cache.countLimit = 200
574	        cache.totalCostLimit = 200 * 192 * 192 * 4
575	        return cache
576	    }()
577	    /// Its own on-disk URL cache (small in memory), so art evicted from `images` or from a
578	    /// previous launch comes back from disk instead of the network.
579	    nonisolated private static let session: URLSession = {
580	        let config = URLSessionConfiguration.default
581	        config.urlCache = URLCache(memoryCapacity: 2 * 1024 * 1024, diskCapacity: 100 * 1024 * 1024,
582	                                   directory: FileManager.default.urls(for: .cachesDirectory, in: .userDomainMask).first?
583	                                       .appendingPathComponent(Bundle.main.bundleIdentifier ?? "StarPlayrX", isDirectory: true)
584	                                       .appendingPathComponent("ArtCache", isDirectory: true))
585	        config.requestCachePolicy = .returnCacheDataElseLoad   // art at a URL never changes
586	        return URLSession(configuration: config)
587	    }()
588	    // Download and decode run off the main actor so arriving art never stalls scrolling;
589	    // the NSImage is made here on the main actor, since NSImage is only Sendable from macOS 14.
590	    private var loading: [String: Task<Thumbnail?, Never>] = [:]
591	    /// `sized(_:pixels:)` results: it base64-decodes JSON, too slow to redo in every row's body.
592	    private var sizedURLs: [String: String] = [:]
593	
594	    /// A decoded bitmap handed back from a background task.
595	    struct Thumbnail: @unchecked Sendable { let image: CGImage }
596	
597	    /// Pixels for art shown at `points`: 3x, so square logos enlarged 1.12x to fill the squircle
598	    /// are still downscaled on a 2x display instead of stretched (which made them blurry).
599	    static let scale: CGFloat = 3
600	    static func pixels(_ points: CGFloat) -> Int { Int((points * scale).rounded(.up)) }
601	
602	    // Keyed by pixels too: hosts with one size give the 32pt row and the 64pt header the same URL,
603	    // and the header must not get the row's small bitmap.
604	    func cached(_ url: String, pixels: Int) -> NSImage? {
605	        images.object(forKey: "\(pixels) \(sizedURL(url, pixels: pixels))" as NSString)
606	    }
607	
608	    func image(_ url: String, pixels: Int) async -> NSImage? {
609	        let url = sizedURL(url, pixels: pixels)
610	        let key = "\(pixels) \(url)"
611	        if let image = images.object(forKey: key as NSString) { return image }
612	        guard !url.isEmpty, let address = URL(string: url) else { return nil }
613	        let task = loading[key] ?? Task.detached(priority: .utility) {
614	            guard let data = try? await ArtCache.session.data(from: address).0,
615	                  let thumb = ArtCache.thumbnail(data, pixels: pixels) else { return nil }
616	            return Thumbnail(image: thumb)
617	        }
618	        loading[key] = task
619	        let thumb = await task.value?.image
620	        loading[key] = nil
621	        if let image = images.object(forKey: key as NSString) { return image }   // another caller already made it
622	        guard let thumb else { return nil }
623	        let image = NSImage(cgImage: thumb, size: NSSize(width: CGFloat(thumb.width) / Self.scale, height: CGFloat(thumb.height) / Self.scale))
624	        images.setObject(image, forKey: key as NSString, cost: thumb.bytesPerRow * thumb.height)
625	        return image
626	    }
627	
628	    private func sizedURL(_ url: String, pixels: Int) -> String {
629	        let key = "\(pixels) \(url)"
630	        if let sized = sizedURLs[key] { return sized }
631	        let sized = Self.sized(url, pixels: pixels)
632	        sizedURLs[key] = sized
633	        return sized
634	    }
635	
636	    /// The same art at the smallest size the server offers that still covers `pixels`.
637	    /// Other hosts (live channel logos, Xtra track art) only have one size and are just decoded smaller.
638	    private static func sized(_ url: String, pixels: Int) -> String {
639	        // SiriusXM's image server (channel tiles): the resize is part of the base64 JSON path.
640	        let imageServer = "https://imgsrv-sxm-prod-device.streaming.siriusxm.com/"
641	        if url.hasPrefix(imageServer),
642	           let data = Data(base64Encoded: String(url.dropFirst(imageServer.count))),
643	           var spec = (try? JSONSerialization.jsonObject(with: data)) as? [String: Any] {
644	            spec["edits"] = [["resize": ["width": pixels, "height": pixels]]]
645	            guard let json = try? JSONSerialization.data(withJSONObject: spec, options: [.sortedKeys, .withoutEscapingSlashes])
646	            else { return url }
647	            return imageServer + json.base64EncodedString()
648	        }
649	        // Live album art comes as _m (450px); _t is 75px and _s 170px.
650	        if url.hasPrefix("http://albumart.siriusxm.com/"), url.hasSuffix("_m.jpg") {
651	            let size = pixels <= 75 ? "_t" : pixels <= 170 ? "_s" : "_m"
652	            return url.dropLast("_m.jpg".count) + size + ".jpg"
653	        }
654	        return url
655	    }
656	
657	    /// Decodes straight to a small bitmap, so neither the downloaded Data nor a full-size image is kept.
658	    nonisolated private static func thumbnail(_ data: Data, pixels: Int) -> CGImage? {
659	        guard let source = CGImageSourceCreateWithData(data as CFData, nil) else { return nil }
660	        let options: [CFString: Any] = [kCGImageSourceCreateThumbnailFromImageAlways: true,
661	                                        kCGImageSourceCreateThumbnailWithTransform: true,
662	                                        kCGImageSourceShouldCacheImmediately: true,
663	                                        kCGImageSourceThumbnailMaxPixelSize: pixels]
664	        return CGImageSourceCreateThumbnailAtIndex(source, 0, options as CFDictionary)
665	    }
666	}

668	/// Shows the image at `url` from `ArtCache`, loading it lazily at 2x `size` points. `content` gets
669	/// the image and whether it is album art (`isAlbumArt` belongs to `url`); nil when nothing is loaded.
670	private struct CachedArt<Content: View>: View {
671	    let url: String
672	    let size: CGFloat
673	    var isAlbumArt = false
674	    @ViewBuilder let content: (NSImage?, Bool) -> Content
675	    @State private var loaded: (url: String, image: NSImage?, isAlbumArt: Bool)?
676	
677	    var body: some View {
678	        let pixels = ArtCache.pixels(size)
679	        let image = loaded?.url == url ? loaded?.image : ArtCache.shared.cached(url, pixels: pixels)
680	        // Until the new art arrives, keep showing the previous art as it was (album art stays album
681	        // art): no flash of the logo or the StarPlayrX icon in between, in the header or in a row.
682	        Group {
683	            if let image { content(image, isAlbumArt) } else { content(loaded?.image, loaded?.isAlbumArt ?? false) }
684	        }
685	        .task(id: url) {
686	            let image = await ArtCache.shared.image(url, pixels: pixels)
687	            if !Task.isCancelled { loaded = (url, image, isAlbumArt) }
688	        }
689	    }
690	}

737	/// One line of text that never truncates: if it doesn't fit, it scrolls smoothly left in a loop
738	/// (pausing at the start of each pass); if it fits, it just sits still.
739	private struct MarqueeText: View {
740	    let text: String
741	    let font: Font
742	    @State private var textWidth: CGFloat = 0
743	    @State private var start = Date()
744	
745	    private let gap: CGFloat = 40      // space between the end of the text and its repeat
746	    private let speed: CGFloat = 30    // points per second
747	    private let pause: Double = 2      // seconds held at the start of each pass
748	
749	    var body: some View {
750	        // Invisible line sets the height; the width comes from the parent.
751	        Text(" ").font(font).hidden()
752	            .frame(maxWidth: .infinity, alignment: .leading)
753	            .background(alignment: .leading) {
754	                Text(text).font(font).fixedSize().hidden()
755	                    .background(GeometryReader { g in
756	                        Color.clear.preference(key: MarqueeWidthKey.self, value: g.size.width)
757	                    })
758	            }
759	            .onPreferenceChange(MarqueeWidthKey.self) { textWidth = $0 }
760	            .overlay(alignment: .leading) {
761	                GeometryReader { geo in
762	                    if textWidth <= geo.size.width {
763	                        Text(text).font(font).fixedSize()
764	                    } else {
765	                        TimelineView(.animation) { ctx in
766	                            let cycle = textWidth + gap
767	                            let period = Double(cycle / speed) + pause
768	                            let t = ctx.date.timeIntervalSince(start).truncatingRemainder(dividingBy: period)
769	                            HStack(spacing: gap) {
770	                                Text(text).font(font).fixedSize()
771	                                Text(text).font(font).fixedSize()
772	                            }
773	                            .offset(x: -CGFloat(max(0, t - pause)) * speed)
774	                        }
775	                    }
776	                }
777	                .clipped()
778	            }
779	    }
780	}
781	
782	private struct MarqueeWidthKey: PreferenceKey {
783	    static var defaultValue: CGFloat = 0
784	    static func reduce(value: inout CGFloat, nextValue: () -> CGFloat) { value = max(value, nextValue()) }
785	}
