247	        }

249	            let screen = statusItem.button?.window?.screen?.visibleFrame ?? .infinite

251	            let top = NSPoint(x: x, y: button.minY - 4)
252	            if panel.isVisible {
253	                // Torn off: glide back under the menu bar instead of jumping. Stepped by hand
254	                // (top-left point per frame) so the unshade resize can't cut the animation short.
255	                let start = NSPoint(x: panel.frame.minX, y: panel.frame.maxY)
256	                let began = CACurrentMediaTime()
257	                let duration = 0.4
258	                glide?.invalidate()

261	                        guard let self else { timer.invalidate(); return }
262	                        let t = min((CACurrentMediaTime() - began) / duration, 1)
263	                        let eased = t < 0.5 ? 2 * t * t : 1 - pow(-2 * t + 2, 2) / 2   // ease in-out
264	                        self.panel.setFrameTopLeftPoint(NSPoint(x: start.x + (top.x - start.x) * eased,
265	                                                                y: start.y + (top.y - start.y) * eased))
266	                        guard t >= 1 else { return }
267	                        timer.invalidate()
268	                        self.glide = nil
269	                        self.panel.refreshShadow()
270	                        if unshade {
271	                            // Same as the double-click unshade: the panel grows from its pinned top edge.
272	                            self.detached = false
273	                            self.shaded = false
274	                        }

276	                }
277	            } else {

552	    func image(_ url: String) async -> NSImage? {
553	        if let image = cached(url) { return image }
554	        guard !url.isEmpty, let address = URL(string: url) else { return nil }
555	        let task = loading[url] ?? Task<Data?, Never> {
556	            try? await URLSession.shared.data(from: address).0
557	        }
558	        loading[url] = task
559	        let data = await task.value
560	        loading[url] = nil
561	        if let image = cached(url) { return image }   // another caller already decoded it
562	        guard let data, let image = NSImage(data: data) else { return nil }
563	        images.setObject(image, forKey: url as NSString)
564	        return image
565	    }
566	}

568	/// Shows the image at `url` from `ArtCache`, loading it lazily; `content` gets nil while it is
569	/// missing, loading or failed.
570	private struct CachedArt<Content: View>: View {
571	    let url: String
572	    @ViewBuilder let content: (NSImage?) -> Content
573	    @State private var loaded: (url: String, image: NSImage?)?
574	
575	    var body: some View {
576	        content(loaded?.url == url ? loaded?.image : ArtCache.shared.cached(url))
577	            .task(id: url) {
578	                let image = await ArtCache.shared.image(url)
579	                if !Task.isCancelled { loaded = (url, image) }
580	            }
581	    }

584	/// The channel search box. A plain AppKit search field with AutoFill, text completion and inline
585	/// predictions turned off: SwiftUI's TextField sometimes popped up an empty AutoFill list.

590	    func makeNSView(context: Context) -> NSSearchField {
591	        let field = NSSearchField()

593	        field.contentType = nil
594	        field.isAutomaticTextCompletionEnabled = false
595	        field.sendsSearchStringImmediately = true
596	        field.sendsWholeSearchString = false
597	        field.delegate = context.coordinator
598	        return field
599	    }
600	
601	    func updateNSView(_ field: NSSearchField, context: Context) {
602	        context.coordinator.text = $text
603	        if field.stringValue != text { field.stringValue = text }   // e.g. set from AppleScript
604	    }

607
