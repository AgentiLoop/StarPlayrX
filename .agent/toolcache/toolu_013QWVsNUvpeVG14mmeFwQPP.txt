525	/// Channel logos and album art kept in memory once loaded. Nothing is fetched until it's shown,
526	/// and a URL that is already loading is shared instead of downloaded twice.
527	@MainActor
528	final class ArtCache {
529	    static let shared = ArtCache()
530	    private let images = NSCache<NSString, NSImage>()
531	    private var loading: [String: Task<NSImage?, Never>] = [:]
532	
533	    func cached(_ url: String) -> NSImage? { images.object(forKey: url as NSString) }
534	
535	    func image(_ url: String) async -> NSImage? {
536	        if let image = cached(url) { return image }
537	        if let task = loading[url] { return await task.value }
538	        guard !url.isEmpty, let address = URL(string: url) else { return nil }
539	        let task = Task<NSImage?, Never> {
540	            guard let (data, _) = try? await URLSession.shared.data(from: address) else { return nil }
541	            return NSImage(data: data)
542	        }
543	        loading[url] = task
544	        let image = await task.value
545	        loading[url] = nil
546	        if let image { images.setObject(image, forKey: url as NSString) }
547	        return image
548	    }

567	/// The channel search box. A plain AppKit search field with AutoFill, text completion and inline
568	/// predictions turned off: SwiftUI's TextField sometimes popped up an empty AutoFill list.
569	private struct SearchField: NSViewRepresentable {
570	    @Binding var text: String
571	    let placeholder: String
572	
573	    func makeNSView(context: Context) -> NSSearchField {
574	        let field = NSSearchField()
575	        field.placeholderString = placeholder
576	        field.contentType = nil
577	        field.isAutomaticTextCompletionEnabled = false
578	        field.sendsSearchStringImmediately = true
579	        field.sendsWholeSearchString = false
580	        field.delegate = context.coordinator
581	        return field
582	    }
583	
584	    func updateNSView(_ field: NSSearchField, context: Context) {
585	        context.coordinator.text = $text
586	        if field.stringValue != text { field.stringValue = text }   // e.g. set from AppleScript
587	    }
588	
589	    func makeCoordinator() -> Coordinator { Coordinator(text: $text) }
590	
591	    final class Coordinator: NSObject, NSSearchFieldDelegate {
592	        var text: Binding<String>
593	        init(text: Binding<String>) { self.text = text }
594	
595	        func controlTextDidBeginEditing(_ note: Notification) {
596	            guard let editor = note.userInfo?["NSFieldEditor"] as? NSTextView else { return }
597	            editor.isAutomaticTextCompletionEnabled = false
598	            if #available(macOS 14.0, *) { editor.inlinePredictionType = .no }
599	        }
600	
601	        func controlTextDidChange(_ note: Notification) {
602	            guard let field = note.object as? NSTextField else { return }
603	            text.wrappedValue = field.stringValue
604	        }
605	    }
606	}
