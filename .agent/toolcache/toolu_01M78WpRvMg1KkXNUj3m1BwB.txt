555	    private let images = NSCache<NSString, NSImage>()
556	    // Downloads hand back Data (Sendable); the NSImage is made here on the main actor, since
557	    // NSImage is only Sendable from macOS 14 and the app supports macOS 13.
558	    private var loading: [String: Task<Data?, Never>] = [:]
559	
560	    func cached(_ url: String) -> NSImage? { images.object(forKey: url as NSString) }
561	
562	    func image(_ url: String) async -> NSImage? {
563	        if let image = cached(url) { return image }
564	        guard !url.isEmpty, let address = URL(string: url) else { return nil }
565	        let task = loading[url] ?? Task<Data?, Never> {
566	            try? await URLSession.shared.data(from: address).0
567	        }
568	        loading[url] = task
569	        let data = await task.value
570	        loading[url] = nil
571	        if let image = cached(url) { return image }   // another caller already decoded it
572	        guard let data, let image = NSImage(data: data) else { return nil }
573	        images.setObject(image, forKey: url as NSString)
574	        return image
575	    }
576	}

578	/// Shows the image at `url` from `ArtCache`, loading it lazily; `content` gets nil while it is
579	/// missing, loading or failed.
580	private struct CachedArt<Content: View>: View {
581	    let url: String
582	    @ViewBuilder let content: (NSImage?) -> Content
583	    @State private var loaded: (url: String, image: NSImage?)?
584	
585	    var body: some View {
586	        content(loaded?.url == url ? loaded?.image : ArtCache.shared.cached(url))
587	            .task(id: url) {
588	                let image = await ArtCache.shared.image(url)
589	                if !Task.isCancelled { loaded = (url, image) }
590	            }
591	    }
592	}

594	/// The channel search box. A plain AppKit search field with AutoFill, text completion and inline
595	/// predictions turned off: SwiftUI's TextField sometimes popped up an empty AutoFill list.
596	private struct SearchField: NSViewRepresentable {
597	    @Binding var text: String
598	    let placeholder: String
599	
600	    func makeNSView(context: Context) -> NSSearchField {
601	        let field = NSSearchField()
602	        field.placeholderString = placeholder
603	        field.contentType = nil
604	        field.isAutomaticTextCompletionEnabled = false
605	        field.sendsSearchStringImmediately = true
606	        field.sendsWholeSearchString = false
607	        field.delegate = context.coordinator
608	        return field
609	    }
610	
611	    func updateNSView(_ field: NSSearchField, context: Context) {
612	        context.coordinator.text = $text
613	        if field.stringValue != text { field.stringValue = text }   // e.g. set from AppleScript
614	    }
615	
616	    func makeCoordinator() -> Coordinator { Coordinator(text: $text) }
617	
618	    final class Coordinator: NSObject, NSSearchFieldDelegate {
619	        var text: Binding<String>
620	        init(text: Binding<String>) { self.text = text }
621	
622	        func controlTextDidBeginEditing(_ note: Notification) {
623	            guard let editor = note.userInfo?["NSFieldEditor"] as? NSTextView else { return }
624	            editor.isAutomaticTextCompletionEnabled = false
625	            if #available(macOS 14.0, *) { editor.inlinePredictionType = .no }
626	        }
627	
628	        func controlTextDidChange(_ note: Notification) {
629	            guard let field = note.object as? NSTextField else { return }
630	            text.wrappedValue = field.stringValue
631	        }
