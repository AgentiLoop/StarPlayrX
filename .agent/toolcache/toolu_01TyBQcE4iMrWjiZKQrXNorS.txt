554	    func cached(_ url: String) -> NSImage? { images.object(forKey: url as NSString) }
555	
556	    func image(_ url: String) async -> NSImage? {
557	        if let image = cached(url) { return image }
558	        guard !url.isEmpty, let address = URL(string: url) else { return nil }
559	        let task = loading[url] ?? Task<Data?, Never> {
560	            try? await URLSession.shared.data(from: address).0
561	        }
562	        loading[url] = task
563	        let data = await task.value
564	        loading[url] = nil
565	        if let image = cached(url) { return image }   // another caller already decoded it
566	        guard let data, let image = NSImage(data: data) else { return nil }
567	        images.setObject(image, forKey: url as NSString)
568	        return image
569	    }
570	}

572	/// Shows the image at `url` from `ArtCache`, loading it lazily; `content` gets nil while it is
573	/// missing, loading or failed.
574	private struct CachedArt<Content: View>: View {
575	    let url: String
576	    @ViewBuilder let content: (NSImage?) -> Content
577	    @State private var loaded: (url: String, image: NSImage?)?
578	
579	    var body: some View {
580	        content(loaded?.url == url ? loaded?.image : ArtCache.shared.cached(url))
581	            .task(id: url) {
582	                let image = await ArtCache.shared.image(url)
583	                if !Task.isCancelled { loaded = (url, image) }
584	            }
585	    }
586	}
