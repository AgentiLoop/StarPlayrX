//MARK: Swifter Embedded Web Server Routes
public func streamingServer() -> HttpServer {
    let server = HttpServer()
    
    autoreleasepool {
        server.get ["/api/v3/ping"]           = pingRoute(pong: "pong")
        server.get ["/api/v3/us"]             = regionRoute(region: "us")
        server.get ["/api/v3/ca"]             = regionRoute(region: "ca")
        server.post["/api/v3/login"]          = loginRoute()
        server.post["/api/v3/session"]        = sessionRoute()
        server.post["/api/v3/channels"]       = channelsRoute()
        server.get ["/api/v3/pdt"]            = pdtRoute()
        server.get ["/api/v3/key"]            = keyOneRoute()
        server.get ["/api/v3/m3u/:channelid"] = playlistRoute()
        server.get ["/api/v3/aac/:aac"]       = audioRoute()
        server.get ["/api/v3/routes"]         = checkRoute(server: server)
    }

    return server
}

    84	        isBusy = true
    85	        channelsLoaded = false
    86	        defer { isBusy = false }
    87	        status = "Starting…"

    93	        // StarPlayrRadioKit caches the password in UserDefaults; the Mac app keeps it only in the Keychain.

    95	
    96	        guard let result, result["success"] as? Bool == true else {
    97	            status = (result?["message"] as? String) ?? "Login failed. Check your connection."
    98	            return false
    99	        }
   100	        Keychain.save(user: user, pass: pass)
   101	
   102	        status = "Loading channels…"
   103	        _ = await post("session", ["channelid": "siriushits1"])
   104	        guard let data = await post("channels", ["channeltype": ""])?["data"] as? [String: Any], !data.isEmpty else {
   105	            status = "Error reading channels."
   106	            return false
   107	        }
   108	
   109	        channels = data.compactMap { key, value in
   110	            guard let v = value as? [String: Any],
   111	                  let number = v["channelNumber"] as? String,
   112	                  var name = v["name"] as? String else { return nil }
   113	            for prefix in ["SiriusXM ", "Sirius XM ", "Sirius ", "SXM ", "SPX "] {
   114	                name = name.replacingOccurrences(of: prefix, with: "")
   115	            }
   116	            return Channel(id: number, name: name,
   117	                           category: v["category"] as? String ?? "",
   118	                           imageURL: v["mediumImage"] as? String ?? "")
   119	        }

   123	        status = "Signed in"

   125	            current = channels.first { $0.id == last }
   126	            // Auto play the last channel played (Xtra channels auto play once they load below).
   127	            if let current, !isPlaying { play(current) }
   128	        }

   135	    /// Xtra channels come from SiriusXM's newer API; if that fails the live channels still work.
   136	    private func loadXtraChannels(user: String, pass: String) async {
   137	        defer { if isLoggedIn { channelsLoaded = true } }
   138	        do {
   139	            try await xtra.signIn(user: user, pass: pass)
   140	            let all = try await xtra.channels()
   141	            let xtras = all.filter(\.isXtra).map {
   142	                Channel(id: $0.number, name: $0.name, category: $0.genre, imageURL: $0.imageURL, xtraID: $0.id)
   143	            }
   144	            // Live channels get the same full square tile art as Xtra channels when the newer API has it.
   145	            var tiles = [String: String]()
   146	            for ch in all where !ch.isXtra && !ch.imageURL.isEmpty { tiles[ch.number] = ch.imageURL }
   147	            guard isLoggedIn else { return }
   148	            liveIDs = Dictionary(all.filter { !$0.isXtra }.map { ($0.number, $0.id) }) { first, _ in first }
   149	            let withTiles = channels.map { ch in
   150	                tiles[ch.id].map { Channel(id: ch.id, name: ch.name, category: ch.category, imageURL: $0, xtraID: ch.xtraID) } ?? ch
   151	            }
   152	            let live = Set(withTiles.map(\.id))
   153	            channels = (withTiles + xtras.filter { !live.contains($0.id) })
   154	                .sorted { $0.id.localizedStandardCompare($1.id) == .orderedAscending }
   155	            let wasNil = current == nil
   156	            if let id = current?.id ?? UserDefaults.standard.string(forKey: "lastChannel") {
   157	                current = channels.first { $0.id == id }   // pick up the new art for the current channel too
   158	                // Auto play a last-played Xtra channel, which wasn't available at login.
   159	                if wasNil, let current, current.xtraID != nil, !isPlaying { play(current) }
   160	            }
   161	            await updatePDT()   // live rows from lookAround right away, not on the next 5 s tick
   162	        } catch {
   163	            print("Xtra channels unavailable: \(error.localizedDescription)")
   164	        }
   165	    }

   188	    func play(_ channel: Channel) {
   189	        haltXtra()
   190	        current = channel
   191	        UserDefaults.standard.set(channel.id, forKey: "lastChannel")
   192	        if let xtraID = channel.xtraID {
   193	            player.pause()
   194	            player.replaceCurrentItem(with: nil)
   195	            playXtra(channel, id: xtraID)
   196	            return
   197	        }
   198	        Task {
   199	            await startServerIfNeeded()
   200	            guard let url = URL(string: "http://localhost:\(port)/api/v3/m3u/\(channel.id).m3u8") else { return }
   201	            let item = AVPlayerItem(url: url)
   202	            // Start as soon as a little audio is buffered instead of waiting to fill a large buffer.
   203	            item.preferredForwardBufferDuration = 3
   204	            // Stay paused until the start position is settled so the old position never plays
   205	            // (that was the half-second stutter before the jump).
   206	            player.pause()
   207	            player.replaceCurrentItem(with: item)
   208	            isPlaying = true
   209	            var tries = 0
   210	            while (item.status != .readyToPlay || item.seekableTimeRanges.isEmpty), tries < 100 {
   211	                try? await Task.sleep(nanoseconds: 100_000_000)
   212	                tries += 1
   213	            }
   214	            guard player.currentItem === item else { return }
   215	            // Fallback: only jump if the start tag was ignored and we're more than 1 s off live.
   216	            if let range = item.seekableTimeRanges.last?.timeRangeValue {
   217	                let edge = max(CMTimeSubtract(CMTimeRangeGetEnd(range), CMTime(seconds: 1, preferredTimescale: 1000)), range.start)
   218	                if abs(CMTimeGetSeconds(CMTimeSubtract(edge, item.currentTime()))) > 1 {
   219	                    // Exact seek so we land within 1 s of live, not on an earlier segment boundary.
   220	                    await item.seek(to: edge, toleranceBefore: .zero, toleranceAfter: .zero)
   221	                }
   222	            }
   223	            guard player.currentItem === item else { return }
   224	            player.play()
   225	            await updatePDT()
   226	        }
   227	    }

   243	    private func playXtra(_ channel: Channel, id: String) {
   244	        isPlaying = true
   245	        // Tuning again returns a different track than the peek, so start with the peeked track
   246	        // the channel list is showing (from its beginning), then continue with the tuned queue.
   247	        let peeked = xtraPeekedTrack.removeValue(forKey: channel.id).map {
   248	            XtraTrack(id: $0.id, artist: $0.artist, title: $0.title, url: $0.url, keyID: $0.keyID,
   249	                      imageURL: $0.imageURL, startOffset: 0)
   250	        }
   251	        print("XTRA play \(channel.id) shown=\(nowPlaying[channel.id]?.song ?? "nil") peeked=\(peeked?.title ?? "nil")")
   252	        if peeked == nil {
   253	            xtraNowPlaying[channel.id] = nil
   254	            nowPlaying[channel.id] = nil
   255	        }
   256	        updateNowPlayingInfo()
   257	        xtraTask = Task {
   258	            var pending = [XtraTrack]()
   259	            var token: String?
   260	            var seen = Set<String>()
   261	            var first = true
   262	            if let peeked {
   263	                seen.insert(peeked.id)
   264	                do {
   265	                    let file = try await xtra.download(peeked, into: xtraFolder)
   266	                    if !Task.isCancelled {
   267	                        let item = AVPlayerItem(url: file)
   268	                        xtraTracks[item] = peeked
   269	                        xtraPlayer.insert(item, after: nil)
   270	                        first = false
   271	                        xtraPlayer.play()
   272	                        xtraItemChanged()
   273	                    }
   274	                } catch {
   275	                    print("XTRA peeked download failed \(channel.id) \(peeked.title): \(error.localizedDescription) url=\(peeked.url)")
   276	                }
   277	            }
   278	            do {
   279	                while !Task.isCancelled {
   280	                    // Keep one track playing and one ready behind it.
   281	                    if xtraPlayer.items().count > 1 {
   282	                        try await Task.sleep(nanoseconds: 2_000_000_000)
   283	                        continue
   284	                    }
   285	                    if pending.isEmpty {
   286	                        let batch = try await xtra.tune(channelID: id, sequenceToken: token)
   287	                        token = batch.token
   288	                        pending = batch.tracks.filter { seen.insert($0.id).inserted }
   289	                        if pending.isEmpty { try await Task.sleep(nanoseconds: 5_000_000_000); continue }
   290	                    }
   291	                    let track = pending.removeFirst()
   292	                    let file = try await xtra.download(track, into: xtraFolder)
   293	                    guard !Task.isCancelled else { break }
   294	                    let item = AVPlayerItem(url: file)
   295	                    xtraTracks[item] = track
   296	                    xtraPlayer.insert(item, after: nil)
   297	                    if first {
   298	                        first = false
   299	                        if track.startOffset > 0 {
   300	                            await xtraPlayer.seek(to: CMTime(seconds: track.startOffset, preferredTimescale: 1000))
   301	                        }
   302	                        xtraPlayer.play()
   303	                        xtraItemChanged()
   304	                    }
   305	                }
   306	            } catch is CancellationError {
   307	            } catch {
   308	                print("Xtra playback error: \(error.localizedDescription)")
   309	                if first, current?.id == channel.id { stop() }
   310	            }
   311	        }
   312	    }

   314	    private func haltXtra() {
   315	        // The list keeps showing the song that was playing, so tuning back in should play that song.
   316	        if let item = xtraPlayer.currentItem, let track = xtraTracks[item], let current, current.xtraID != nil {
   317	            xtraPeekedTrack[current.id] = track
   318	        }
   319	        xtraTask?.cancel()
   320	        xtraTask = nil
   321	        xtraPlayer.pause()
   322	        xtraPlayer.removeAllItems()
   323	        xtraTracks = [:]
   324	        try? FileManager.default.removeItem(at: xtraFolder)
   325	    }

   327	    /// Shows the queued track that just started and deletes files of tracks that have finished.
   328	    fileprivate func xtraItemChanged() {
   329	        let queued = Set(xtraPlayer.items())
   330	        for (item, track) in xtraTracks where !queued.contains(item) {
   331	            xtraTracks[item] = nil
   332	            try? FileManager.default.removeItem(at: xtraFolder.appendingPathComponent("\(track.id).aac"))
   333	        }
   334	        guard let item = xtraPlayer.currentItem, let track = xtraTracks[item], let current, current.xtraID != nil else { return }
   335	        print("XTRA playing \(current.id) \(track.artist) – \(track.title)")
   336	        let np = NowPlaying(artist: track.artist, song: track.title, image: track.imageURL)
   337	        xtraNowPlaying[current.id] = np
   338	        nowPlaying[current.id] = np
   339	        updateNowPlayingInfo()
   340	    }

   351	    func updatePDT() async {
   352	        // One source for every live row, tuned or not: the newer API (lookAround).
   353	        if let live = try? await xtra.liveNowPlaying() {
   354	            for (number, id) in liveIDs {
   355	                guard let la = live[id] else { continue }
   356	                lookAround[number] = Self.keepingArt(la, of: lookAround[number])
   357	            }
   358	        }
   359	        nowPlaying = lookAround.merging(xtraNowPlaying) { live, _ in live }
   360	        updateNowPlayingInfo()
   361	        Task { await refreshXtraNowPlaying() }
   362	    }
   363	    private var lookAround = [String: NowPlaying]()   // last lookAround cut per live channel number
   364	
   365	    /// Xtra channels have no PDT feed, so ask each idle one what it would play if tuned now.
   366	    /// Channels never peeked are all filled at once; after that the stalest are refreshed
   367	    /// a batch at a time on every PDT tick. Each result is shown as soon as it arrives.
   368	    private func refreshXtraNowPlaying() async {
   369	        guard isLoggedIn, !xtraPeeking else { return }
   370	        xtraPeeking = true
   371	        defer { xtraPeeking = false }
   372	        let stale = Date().addingTimeInterval(-180)
   373	        let idle = channels.filter { $0.xtraID != nil && !isPlayingXtra($0.id) }
   374	        let unpeeked = idle.filter { xtraPeekedAt[$0.id] == nil }
   375	        let batch = !unpeeked.isEmpty ? unpeeked : Array(idle
   376	            .filter { xtraPeekedAt[$0.id]! < stale }
   377	            .sorted { xtraPeekedAt[$0.id]! < xtraPeekedAt[$1.id]! }
   378	            .prefix(30))
   379	        guard !batch.isEmpty else { return }
   380	        let client = xtra
   381	        await withTaskGroup(of: (String, XtraTrack?).self) { group in
   382	            for (i, ch) in batch.enumerated() {
   383	                if i >= 12, let r = await group.next() { applyXtraPeek(r.0, r.1) }
   384	                group.addTask { (ch.id, try? await client.peek(channelID: ch.xtraID!)) }
   385	            }
   386	            for await r in group { applyXtraPeek(r.0, r.1) }
   387	        }
   388	    }

   390	    /// Peek results arrive many per second; publishing each one redraws the whole channel
   391	    /// list and stalls scrolling, so they're batched and published at most every 2 seconds.
   392	    private func applyXtraPeek(_ id: String, _ track: XtraTrack?) {
   393	        guard isLoggedIn else { return }
   394	        xtraPeekedAt[id] = Date()
   395	        guard let track, !isPlayingXtra(id) else { return }
   396	        // The played track, xtraNowPlaying and the list text are committed together on flush,
   397	        // so tapping a channel always plays the song its row is showing.
   398	        pendingPeeks[id] = track
   399	        guard !peekFlushScheduled else { return }
   400	        peekFlushScheduled = true
   401	        Task {
   402	            try? await Task.sleep(for: .seconds(2))
   403	            peekFlushScheduled = false
   404	            let tracks = pendingPeeks.filter { !isPlayingXtra($0.key) }
   405	            pendingPeeks.removeAll()
   406	            guard isLoggedIn, !tracks.isEmpty else { return }
   407	            let batch = Dictionary(uniqueKeysWithValues: tracks.map { id, track in
   408	                (id, Self.keepingArt(NowPlaying(artist: track.artist, song: track.title, image: track.imageURL), of: nowPlaying[id]))
   409	            })
   410	            xtraPeekedTrack.merge(tracks) { _, new in new }
   411	            xtraNowPlaying.merge(batch) { _, new in new }
   412	            nowPlaying.merge(batch) { _, new in new }
   413	        }
   414	    }
   415	    /// The PDT feed only has album art for songs the server has just seen on a now-playing-live
   416	    /// fetch (it forgets them on the next one), so a later poll of the same song often comes back
   417	    /// without art. Keep the album art until the song changes, instead of flipping back to the logo.
   418	    nonisolated static func keepingArt(_ new: NowPlaying, of old: NowPlaying?) -> NowPlaying {
   419	        guard new.image.isEmpty, let old, !old.image.isEmpty, old.artist == new.artist, old.song == new.song else { return new }
   420	        var kept = new
   421	        kept.image = old.image
   422	        return kept
   423	    }

   448	    private func updateNowPlayingInfo() {
   449	        let center = MPNowPlayingInfoCenter.default()
   450	        guard let current else { center.nowPlayingInfo = nil; return }
   451	        let np = nowPlaying[current.id]
   452	        var info: [String: Any] = [
   453	            MPMediaItemPropertyTitle: np?.song ?? current.name,
   454	            MPMediaItemPropertyArtist: np?.artist ?? "",
   455	            MPMediaItemPropertyAlbumTitle: current.name,
   456	            MPNowPlayingInfoPropertyIsLiveStream: true
   457	        ]
   458	        // Same art as the menu's Now Playing tile: album art when there is some, else the channel logo.
   459	        let url = np?.image.isEmpty == false ? np!.image : current.imageURL
   460	        if url == artworkURL, let artwork { info[MPMediaItemPropertyArtwork] = artwork }
   461	        center.nowPlayingInfo = info
   462	        center.playbackState = isPlaying ? .playing : .paused
   463	        guard url != artworkURL else { return }
   464	        artworkURL = url
   465	        artwork = nil
   466	        Task {

   493	        guard let url = URL(string: "\(base)/\(path)") else { return nil }
   494	        var req = URLRequest(url: url, cachePolicy: .reloadIgnoringLocalAndRemoteCacheData, timeoutInterval: 2)
   495	        req.httpMethod = "GET"

   500	    private func get(_ path: String) async -> [String: Any]? {
   501	        guard let url = URL(string: "\(base)/\(path)") else { return nil }
   502	        let req = URLRequest(url: url, cachePolicy: .reloadIgnoringLocalAndRemoteCacheData, timeoutInterval: 60)
   503	        guard let (data, _) = try? await URLSession.shared.data(for: req) else { return nil }
   504	        return (try? JSONSerialization.jsonObject(with: data)) as? [String: Any]
   505	    }

   507	    private func post(_ path: String, _ body: [String: Any]) async -> [String: Any]? {
   508	        guard let url = URL(string: "\(base)/\(path)") else { return nil }
   509	        var req = URLRequest(url: url, cachePolicy: .reloadIgnoringLocalAndRemoteCacheData, timeoutInterval: 60)
   510	        req.httpMethod = "POST"
   511	        req.httpBody = try? JSONSerialization.data(withJSONObject: body)
   512	        req.addValue("application/json", forHTTPHeaderField: "Content-Type")
   513	        guard let (data, _) = try? await URLSession.shared.data(for: req) else { return nil }
   514	        return (try? JSONSerialization.jsonObject(with: data)) as? [String: Any]
   515	    }

   517	    // MARK: Port finder
   518	
   519	    private func portIsOpen(_ port: in_port_t) -> Bool {
   520	        let fd = socket(AF_INET, SOCK_STREAM, 0)
   521	        guard fd != -1 else { return false }
   522	        defer { Darwin.shutdown(fd, SHUT_RDWR); close(fd) }
   523	
   524	        var addr = sockaddr_in()
   525	        addr.sin_len = __uint8_t(MemoryLayout<sockaddr_in>.size)
   526	        addr.sin_family = sa_family_t(AF_INET)
   527	        addr.sin_port = port.bigEndian
   528	        addr.sin_addr = in_addr(s_addr: inet_addr("127.0.0.1"))
   529	
   530	        let bound = withUnsafePointer(to: &addr) {
   531	            $0.withMemoryRebound(to: sockaddr.self, capacity: 1) {
   532	                Darwin.bind(fd, $0, socklen_t(MemoryLayout<sockaddr_in>.size))
   533	            }
   534	        }
   535	        return bound != -1 && listen(fd, SOMAXCONN) != -1
   536	    }
