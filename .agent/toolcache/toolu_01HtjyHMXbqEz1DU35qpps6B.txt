210	    /// What every live (non-Xtra) channel is playing right now, keyed by channel entity id, in one
211	    /// unauthenticated call: the "lookAround" feed the web player polls for its channel grid.
212	    /// Talk and sports channels fall back to the current show's art when the cut has none.
213	    /// Done the way the web player does it (its remote config: lookaround pollingMs 45000,
214	    /// livePointIntervalMs 60000): the feed is refetched at most every 45 s. The player runs at live
215	    /// (within 1 s), so the cut on air now is what a listener hears.
216	    func liveNowPlaying() async throws -> [String: NowPlaying] {
217	        if Date().timeIntervalSince(lookAroundFetched) > 45 {
218	            let (data, resp) = try await session.data(from: URL(string: Self.lookAround)!)
219	            if let code = (resp as? HTTPURLResponse)?.statusCode, code >= 400 { throw XtraError.http(code, "lookAround") }
220	            lookAroundChannels = ((try? JSONSerialization.jsonObject(with: data)) as? [String: Any])?["channels"] as? [String: [String: Any]] ?? [:]
221	            lookAroundFetched = Date()
222	        }
223	        let now = Date()
224	        var result = [String: NowPlaying]()
225	        for (id, ch) in lookAroundChannels {
226	            func current(_ key: String) -> [String: Any]? {
227	                (ch[key] as? [[String: Any]] ?? []).last { $0["isAd"] as? Bool != true && (Self.date($0["validFrom"] as? String) ?? now) <= now }
228	            }
229	            guard let cut = current("cuts") else { continue }
230	            let show = current("shows")
231	            let art = ((cut["image"] ?? show?["image"]) as? [String: Any])?["url"] as? String
232	            // Talk cuts have no artist; the web player shows the show's name in its place.
233	            let artist = cut["artistName"] as? String ?? ""
234	            result[id] = NowPlaying(artist: artist.isEmpty ? show?["name"] as? String ?? "" : artist,
235	                                    song: cut["name"] as? String ?? "",
236	                                    image: art.map { Self.imageURL(key: $0, size: 450) } ?? "")
237	        }
