170	/// Spectrum bars whose colours follow the music: each bar drifts at the pace of its own band,
171	/// onsets push it along, and bass kicks throw a fresh random palette. Every colour is checked
172	/// for contrast against the panel. Kicks also make the bars glow and a soft blob swell behind.
173	struct SpectrumView: View {
174	    @ObservedObject private var viz = Visualizer.shared
175	    @Environment(\.colorScheme) private var scheme
176	
177	    private func hue(_ x: Double) -> Double { (x.truncatingRemainder(dividingBy: 1) + 1).truncatingRemainder(dividingBy: 1) }
178	
179	    var body: some View {
180	        let bg = ContrastColor.background(scheme)
181	        HStack(alignment: .bottom, spacing: 2) {
182	            ForEach(viz.levels.indices, id: \.self) { i in
183	                let level = Double(viz.levels[i])
184	                let h = viz.hues[i]
185	                let base = ContrastColor.color(hue: h, saturation: 0.9, against: bg)
186	                let top = ContrastColor.color(hue: hue(h + 0.06 + 0.1 * level), saturation: 0.75, against: bg, minRatio: 4.5)
187	                // A glow in the bar's own colour washes out on a light panel; there it's a deeper shade instead.
188	                let glow = bg < 0.5 ? top : ContrastColor.color(hue: hue(h + 0.06 + 0.1 * level), saturation: 0.9, against: bg, minRatio: 8)
189	                RoundedRectangle(cornerRadius: 1.5)
190	                    .fill(LinearGradient(colors: [base, top], startPoint: .bottom, endPoint: .top))
191	                    .frame(height: max(2, CGFloat(level) * 28))
192	                    .shadow(color: glow.opacity(0.4 + 0.6 * viz.pulse), radius: 1 + 4 * viz.pulse)
193	            }
194	        }
195	        .frame(height: 28, alignment: .bottom)
196	        .background {
197	            // The "lava": a blurred blob of the colour opposite the middle bar's, swelling on the beat.
198	            Ellipse()
199	                .fill(Color(hue: hue(viz.hues[Visualizer.bandCount / 2] + 0.5), saturation: 0.8, brightness: 1))
200	                .scaleEffect(x: 0.6 + 0.3 * sin(viz.phase * 2 * .pi * 2), y: 0.6 + 0.5 * viz.pulse)
201	                .offset(x: 80 * viz.blobX)
202	                .blur(radius: 14)
203	                .opacity(0.075 + 0.175 * viz.pulse)
204	        }
205	        .animation(.linear(duration: 0.05), value: viz.levels)
206	        .onAppear { viz.start() }
207	        .onDisappear { viz.stop() }
208	    }
209	}
