57 	    private func update(_ new: [Float]) {
58 	        guard timer != nil else { return }
59 	        let previous = levels
60 	        // Rise fast but over a few frames, fall slowly. At 60 fps an instant rise showed every
61 	        // frame-to-frame wobble of the FFT as jitter; easing in hides it without adding lag you can see.
62 	        levels = zip(levels, new).map { old, new in new > old ? old + (new - old) * 0.55 : max(new, old * 0.894) }
63 	        rise = rise.indices.map { i in
64 	            (rise[i] + 0.0015 + 0.006 * Double(levels[i]) + 0.004 * pulse).truncatingRemainder(dividingBy: 1)
65 	        }
66 	        // 60 updates a second (every step below is half its 30 fps size): quiet ≈ one colour cycle per 30 s, loud ≈ one per 3 s.
67 	        let energy = Double(new.reduce(0, +) / Float(new.count))
68 	        phase = (phase + 0.0005 + energy * 0.01).truncatingRemainder(dividingBy: 1)
69 	        let bass = new.prefix(4).reduce(0, +) / 4
70 	        let kick = bass - bassAverage
71 	        bassAverage = bassAverage * 0.975 + bass * 0.025
72 	        pulse = max(Double(max(0, kick) * 4).clamped, pulse * 0.922)
73 	
74 	        // Bar colours blend the old rolling wave (global phase plus a slow wave across the bars)
75 	        // halfway toward a random palette. A strong kick (at most every ~3 s) or a rare random
76 	        // moment throws a new palette; each bar glides there at a pace set by its own band,
77 	        // onsets nudge it, and each bar is softened toward its neighbours so colours flow sideways.
78 	        paletteCooldown -= 1
79 	        if (kick > 0.25 && paletteCooldown <= 0) || Int.random(in: 0..<900) == 0 {
80 	            newPalette()
81 	            paletteCooldown = 180
82 	        }
83 	        let t = phase * 2 * .pi
84 	        let glided = hues.indices.map { i in
85 	            let x = Double(i) / Double(Self.bandCount)
86 	            let wave = phase + x * 0.3 + 0.08 * sin(x * 7 + t * 3)
87 	            let target = wave + 0.5 * Self.shortest(hueTargets[i] - wave)
88 	            let level = Double(new[i])
89 	            let onset = Double(max(0, new[i] - previous[i]))
90 	            return hues[i] + Self.shortest(target - hues[i]) * (0.005 + 0.04 * level) + onset * 0.02
91 	        }
92 	        hues = glided.indices.map { i in
93 	            let left = glided[max(0, i - 1)], right = glided[min(glided.count - 1, i + 1)]
94 	            let h = glided[i] + 0.125 * (Self.shortest(left - glided[i]) + Self.shortest(right - glided[i]))
95 	            return (h.truncatingRemainder(dividingBy: 1) + 1).truncatingRemainder(dividingBy: 1)
96 	        }
97 	
98 	        // Spectral centroid, 0 (bass-heavy) ... 1 (treble-heavy).
99 	        let total = new.reduce(0, +)
100	        let weighted = new.enumerated().reduce(Float(0)) { $0 + Float($1.offset) * $1.element }
101	        let centroid = total > 0.01 ? Double(weighted / total) / Double(new.count - 1) : 0.5
102	        // New random spot on a strong kick, or now and then on its own.
103	        if kick > 0.15 || Int.random(in: 0..<180) == 0 { wanderTarget = .random(in: -1...1) }
104	        wander += (wanderTarget - wander) * 0.02
105	        let target = Swift.min(1, Swift.max(-1, (centroid * 2 - 1) * 1.5 + wander * 0.6))
106	        blobX += (target - blobX) * 0.04
107	    }

110	    private static func shortest(_ d: Double) -> Double {
111	        var d = d.truncatingRemainder(dividingBy: 1)
112	        if d > 0.5 { d -= 1 } else if d < -0.5 { d += 1 }
113	        return d

116	    /// Random palette in a random scheme, so the row never settles into one predictable cycle.
117	    private func newPalette() {
118	        let base = Double.random(in: 0..<1)
119	        let scheme = Int.random(in: 0..<5)
120	        let n = Self.bandCount
121	        hueTargets = (0..<n).map { i in
122	            let jitter = Double.random(in: -0.04...0.04)
123	            switch scheme {
124	            case 0: return base + jitter                                          // analogous
125	            case 1: return base + (Bool.random() ? 0 : 0.5) + jitter              // complementary
126	            case 2: return base + Double(Int.random(in: 0..<3)) / 3 + jitter      // triad
127	            case 3: return base + Double(i) / Double(n) * .random(in: 0.3...1)    // sweep
128	            default: return .random(in: 0..<1)                                    // confetti
129	            }
130	        }
131	    }

138	/// Picks bar colours that always stand out from the panel: WCAG relative luminance and
139	/// contrast ratio, lightening (dark mode) or darkening (light mode) a hue only as far as
140	/// needed to reach `minRatio`, so saturated colours survive wherever they already contrast.
141	enum ContrastColor {
142	    /// Approximate luminance of the menu/panel material behind the bars.
143	    static func background(_ scheme: ColorScheme) -> Double { scheme == .dark ? 0.025 : 0.84 }
144	
145	    static func color(hue: Double, saturation: Double, against bg: Double, minRatio: Double = 3.5) -> Color {
146	        let dark = bg < 0.5
147	        // Dark background: full brightness, then wash toward white. Light: darken.
148	        for step in 0...20 {
149	            let k = Double(step) / 20
150	            let s = dark ? saturation * (1 - 0.7 * k) : saturation
151	            let b = dark ? 1 : 1 - 0.75 * k
152	            let (r, g, bl) = rgb(hue, s, b)
153	            let l = 0.2126 * linear(r) + 0.7152 * linear(g) + 0.0722 * linear(bl)
154	            let ratio = (max(l, bg) + 0.05) / (min(l, bg) + 0.05)
155	            if ratio >= minRatio || step == 20 { return Color(red: r, green: g, blue: bl) }
156	        }
157	        return .primary
158	    }
159	
160	    private static func linear(_ c: Double) -> Double {
161	        c <= 0.04045 ? c / 12.92 : pow((c + 0.055) / 1.055, 2.4)
162	    }

164	    private static func rgb(_ h: Double, _ s: Double, _ v: Double) -> (Double, Double, Double) {
165	        let h6 = (h - floor(h)) * 6
166	        let f = h6 - floor(h6)
167	        let p = v * (1 - s), q = v * (1 - s * f), t = v * (1 - s * (1 - f))
168	        switch Int(h6) % 6 {
169	        case 0: return (v, t, p)
170	        case 1: return (q, v, p)
171	        case 2: return (p, v, t)
172	        case 3: return (p, q, v)
173	        case 4: return (t, p, v)
174	        default: return (v, p, q)
175	        }
176	    }

179	/// Spectrum bars whose colours follow the music: each bar drifts at the pace of its own band,
180	/// onsets push it along, and bass kicks throw a fresh random palette. Every colour is checked
181	/// for contrast against the panel. Kicks also make the bars glow and a soft blob swell behind.
182	struct SpectrumView: View {
183	    @ObservedObject private var viz = Visualizer.shared
184	    @Environment(\.colorScheme) private var scheme
185	    /// Luminance of the wallpaper behind the panel (it shows through the translucent material).
186	    @State private var wallpaper: Double?
187	
188	    private func hue(_ x: Double) -> Double { (x.truncatingRemainder(dividingBy: 1) + 1).truncatingRemainder(dividingBy: 1) }
189	
190	    var body: some View {
191	        let fixed = ContrastColor.background(scheme)
192	        // The material mostly hides the wallpaper; let a bright or dark desktop pull the guess part way.
193	        let bg = wallpaper.map { fixed * 0.7 + $0 * 0.3 } ?? fixed
194	        HStack(alignment: .bottom, spacing: 2) {
195	            ForEach(viz.levels.indices, id: \.self) { i in
196	                let level = Double(viz.levels[i])
197	                let h = viz.hues[i]
198	                let base = ContrastColor.color(hue: h, saturation: 0.9, against: fixed)
199	                let top = ContrastColor.color(hue: hue(h + 0.06 + 0.1 * level), saturation: 0.75, against: fixed, minRatio: 4.5)
200	                // A glow in the bar's own colour washes out on a light background; there it's the darkest shade
201	                // of the bar's hue instead, and stronger, so the bars stand off it. The panel is translucent, so a
202	                // bright wallpaper makes it light in Dark Mode too: decide from the wallpaper, not just the scheme.
203	                let light = bg >= 0.5 || (wallpaper ?? 0) >= 0.35
204	                let glow = light ? ContrastColor.color(hue: hue(h + 0.06 + 0.1 * level), saturation: 0.9, against: max(bg, 0.84), minRatio: 14) : top
205	                RoundedRectangle(cornerRadius: 1.5)
206	                    .fill(LinearGradient(colors: [base, top], startPoint: .bottom, endPoint: .top))
207	                    // On a light panel a thin dark edge keeps pale bars from melting into it.
208	                    .overlay {
209	                        if light {
210	                            RoundedRectangle(cornerRadius: 1.5).strokeBorder(.black.opacity(0.25), lineWidth: 0.5)
211	                        }
212	                    }
213	                    .overlay { LavaBubbles(rise: viz.rise[i], seed: Double(i), color: light ? .black : .white) }
214	                    .clipShape(RoundedRectangle(cornerRadius: 1.5))
215	                    .frame(height: max(2, CGFloat(level) * 28))
216	                    .shadow(color: glow.opacity(light ? 0.75 + 0.25 * viz.pulse : 0.4 + 0.6 * viz.pulse), radius: 1 + 4 * viz.pulse)
217	            }
218	        }
219	        .frame(height: 28, alignment: .bottom)
220	        .background {
221	            // The "lava": a blurred blob of the colour opposite the middle bar's, swelling on the beat.
222	            Ellipse()
223	                .fill(Color(hue: hue(viz.hues[Visualizer.bandCount / 2] + 0.5), saturation: 0.8, brightness: 1))
224	                .scaleEffect(x: 0.6 + 0.3 * sin(viz.phase * 2 * .pi * 2), y: 0.6 + 0.5 * viz.pulse)
225	                .offset(x: 80 * viz.blobX)
226	                .blur(radius: 14)
227	                .opacity(0.075 + 0.175 * viz.pulse)
228	        }
229	        .animation(.linear(duration: 1.0 / 60), value: viz.levels)
230	        .background(WallpaperProbe(luminance: $wallpaper))
231	        .onAppear { viz.start() }
232	        .onDisappear { viz.stop() }
233	    }
234	}

237	/// stretch as they climb, swell in the middle and fade near the top; a blur melts touching blobs together.
238	private struct LavaBubbles: View {
239	    let rise: Double
240	    let seed: Double
241	    let color: Color
242	
243	    var body: some View {
244	        Canvas { ctx, size in
245	            guard size.height > 6 else { return }
246	            ctx.addFilter(.blur(radius: 1))
247	            for k in 0..<3 {
248	                let p = (rise + Double(k) / 3 + seed * 0.137).truncatingRemainder(dividingBy: 1)
249	                let d = size.width * (0.45 + 0.25 * sin(p * .pi))            // grow mid-way, shrink at the ends
250	                let x = size.width / 2 + size.width * 0.18 * sin(p * 9 + seed * 2 + Double(k))
251	                let y = size.height * (1.1 - 1.2 * p)                         // from just below the bar to just above
252	                let rect = CGRect(x: x - d / 2, y: y - d * 0.65, width: d, height: d * 1.3)
253	                ctx.fill(Path(ellipseIn: rect), with: .color(color.opacity(0.32 * sin(p * .pi))))

300	        /// Average luminance of the part of the wallpaper (scaled to fill the screen) under `rect`.
301	        private static func luminance(of url: URL, in rect: CGRect, screen: CGRect) -> Double? {
302	            guard let src = CGImageSourceCreateWithURL(url as CFURL, nil),
303	                  let image = CGImageSourceCreateThumbnailAtIndex(src, 0, [kCGImageSourceCreateThumbnailFromImageAlways: true,
304	                                                                          kCGImageSourceThumbnailMaxPixelSize: 512] as CFDictionary)
305	            else { return nil }
306	            let iw = Double(image.width), ih = Double(image.height)
307	            let scale = max(iw / screen.width, ih / screen.height)          // fill: crop the overflow, centred
308	            let ox = (iw - screen.width * scale) / 2, oy = (ih - screen.height * scale) / 2
309	            // Screen y goes up from the bottom; image y goes down from the top.
310	            let x = ox + (rect.minX - screen.minX) * scale
311	            let y = oy + (screen.maxY - rect.maxY) * scale
312	            let crop = CGRect(x: x, y: y, width: max(1, rect.width * scale), height: max(1, rect.height * scale)).integral
313	            guard let part = image.cropping(to: crop) else { return nil }
314	            guard let ctx = CGContext(data: nil, width: 1, height: 1, bitsPerComponent: 8, bytesPerRow: 4,
315	                                      space: CGColorSpace(name: CGColorSpace.sRGB)!,
316	                                      bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return nil }
317	            ctx.interpolationQuality = .medium
318	            ctx.draw(part, in: CGRect(x: 0, y: 0, width: 1, height: 1))

320	            let px = data.assumingMemoryBound(to: UInt8.self)
