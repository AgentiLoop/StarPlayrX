321	    /// Icon click: open under the menu bar, close if open there, or move a torn-off panel back under it.
322	    @objc func statusItemClicked() {
323	        if panel.isVisible && !detached { hide(); return }
324	        // Shaded and torn off: glide back still rolled up, then unroll once in place.
325	        let unshade = panel.isVisible && detached && shaded
326	        if !unshade {
327	            detached = false
328	            shaded = false
329	        }
330	        if let button = statusItem.button?.window?.frame {
331	            let screen = statusItem.button?.window?.screen?.visibleFrame ?? .infinite
332	            let x = min(button.minX, screen.maxX - panel.frame.width)
333	            let top = NSPoint(x: x, y: button.minY - 4)
334	            if panel.isVisible {
335	                // Torn off: glide back under the menu bar instead of jumping. Stepped by hand
336	                // (top-left point per frame) so the unshade resize can't cut the animation short.
337	                let start = NSPoint(x: panel.frame.minX, y: panel.frame.maxY)
338	                let began = CACurrentMediaTime()
339	                let duration = 0.4
340	                glide?.invalidate()
341	                glide = Timer.scheduledTimer(withTimeInterval: 1.0 / 60.0, repeats: true) { [weak self] timer in
342	                    Task { @MainActor in
343	                        guard let self else { timer.invalidate(); return }
344	                        let t = min((CACurrentMediaTime() - began) / duration, 1)
345	                        let eased = t < 0.5 ? 2 * t * t : 1 - pow(-2 * t + 2, 2) / 2   // ease in-out
346	                        self.panel.setFrameTopLeftPoint(NSPoint(x: start.x + (top.x - start.x) * eased,
347	                                                                y: start.y + (top.y - start.y) * eased))
348	                        guard t >= 1 else { return }
349	                        timer.invalidate()
350	                        self.glide = nil
351	                        self.panel.refreshShadow()
352	                        if unshade {
353	                            // Same as the double-click unshade: the panel grows from its pinned top edge.
354	                            self.detached = false
355	                            self.shaded = false
356	                        }
357	                    }
358	                }
359	            } else {
360	                panel.setFrameTopLeftPoint(top)
361	            }
362	        }
363	        // Like MenuBarExtra: the app becomes active (controls draw active, not greyed) and
364	        // the menu bar icon stays highlighted while the panel hangs under it.
365	        NSApp.activate(ignoringOtherApps: true)
366	        statusItem.button?.highlight(true)
367	        panel.makeKeyAndOrderFront(nil)
368	        // Under the menu bar it behaves like a menu: a click in another app closes it.
369	        // Clicks on our own icon are left to statusItemClicked, or the two would fight
370	        // (monitor hides, then the click reopens).
371	        if outsideClick == nil {
372	            outsideClick = NSEvent.addGlobalMonitorForEvents(matching: [.leftMouseDown, .rightMouseDown]) { [weak self] _ in
373	                let point = NSEvent.mouseLocation
374	                Task { @MainActor in
375	                    guard let self, !(self.statusItem.button?.window?.frame.contains(point) ?? false) else { return }
376	                    self.hide()
377	                }
378	            }
379	        }
380	    }

634	/// Channel logos and album art kept in memory once loaded. Nothing is fetched until it's shown or about to be (list look-ahead),
635	/// and a URL that is already loading is shared instead of downloaded twice. Each image is fetched
636	/// (where the server can resize) and decoded at 2x the points it is shown at, so the 32pt channel
637	/// rows and the 64pt Now Playing tile each get their own size, and the cache is bounded.
638	@MainActor
639	final class ArtCache {
640	    static let shared = ArtCache()
641	    private let images: NSCache<NSString, NSImage> = {
642	        let cache = NSCache<NSString, NSImage>()
643	        cache.countLimit = 200
644	        cache.totalCostLimit = 200 * 192 * 192 * 4
645	        return cache
646	    }()
647	    /// Its own on-disk URL cache (small in memory), so art evicted from `images` or from a
648	    /// previous launch comes back from disk instead of the network.
649	    nonisolated private static let session: URLSession = {
650	        let config = URLSessionConfiguration.default
651	        config.urlCache = URLCache(memoryCapacity: 2 * 1024 * 1024, diskCapacity: 100 * 1024 * 1024,
652	                                   directory: FileManager.default.urls(for: .cachesDirectory, in: .userDomainMask).first?
653	                                       .appendingPathComponent(Bundle.main.bundleIdentifier ?? "StarPlayrX", isDirectory: true)
654	                                       .appendingPathComponent("ArtCache", isDirectory: true))
655	        config.requestCachePolicy = .returnCacheDataElseLoad   // art at a URL never changes
656	        return URLSession(configuration: config)
657	    }()
658	    // Download and decode run off the main actor so arriving art never stalls scrolling;
659	    // the NSImage is made here on the main actor, since NSImage is only Sendable from macOS 14.
660	    private var loading: [String: Task<Thumbnail?, Never>] = [:]
661	    /// `sized(_:pixels:)` results: it base64-decodes JSON, too slow to redo in every row's body.
662	    private var sizedURLs: [String: String] = [:]
663	
664	    /// A decoded bitmap handed back from a background task.
665	    struct Thumbnail: @unchecked Sendable { let image: CGImage }
666	
667	    /// Pixels for art shown at `points`: 3x, so square logos enlarged 1.12x to fill the squircle
668	    /// are still downscaled on a 2x display instead of stretched (which made them blurry).
669	    static let scale: CGFloat = 3
670	    static func pixels(_ points: CGFloat) -> Int { Int((points * scale).rounded(.up)) }
671	
672	    // Keyed by pixels too: hosts with one size give the 32pt row and the 64pt header the same URL,
673	    // and the header must not get the row's small bitmap.
674	    func cached(_ url: String, pixels: Int) -> NSImage? {
675	        images.object(forKey: "\(pixels) \(sizedURL(url, pixels: pixels))" as NSString)
676	    }
677	
678	    func image(_ url: String, pixels: Int) async -> NSImage? {
679	        let url = sizedURL(url, pixels: pixels)
680	        let key = "\(pixels) \(url)"
681	        if let image = images.object(forKey: key as NSString) { return image }
682	        guard !url.isEmpty, let address = URL(string: url) else { return nil }
683	        let task = loading[key] ?? Task.detached(priority: .utility) {
684	            guard let data = try? await ArtCache.session.data(from: address).0,
685	                  let thumb = ArtCache.thumbnail(data, pixels: pixels) else { return nil }
686	            return Thumbnail(image: thumb)
687	        }
688	        loading[key] = task
689	        let thumb = await task.value?.image
690	        loading[key] = nil
691	        if let image = images.object(forKey: key as NSString) { return image }   // another caller already made it
692	        guard let thumb else { return nil }
693	        let image = NSImage(cgImage: thumb, size: NSSize(width: CGFloat(thumb.width) / Self.scale, height: CGFloat(thumb.height) / Self.scale))
694	        images.setObject(image, forKey: key as NSString, cost: thumb.bytesPerRow * thumb.height)
695	        return image
696	    }
697	
698	    /// Starts loading art that isn't shown yet; `image` shares the download if a row asks for it meanwhile.
699	    func prefetch(_ url: String, pixels: Int) {
700	        guard !url.isEmpty, cached(url, pixels: pixels) == nil else { return }
701	        Task { _ = await image(url, pixels: pixels) }
702	    }
703	
704	    private func sizedURL(_ url: String, pixels: Int) -> String {
705	        let key = "\(pixels) \(url)"
706	        if let sized = sizedURLs[key] { return sized }
707	        let sized = Self.sized(url, pixels: pixels)
708	        sizedURLs[key] = sized
709	        return sized
710	    }
711	
712	    /// The same art at the smallest size the server offers that still covers `pixels`.
713	    /// Other hosts (live channel logos, Xtra track art) only have one size and are just decoded smaller.
714	    private static func sized(_ url: String, pixels: Int) -> String {
715	        // SiriusXM's image server (channel tiles): the resize is part of the base64 JSON path.
716	        let imageServer = "https://imgsrv-sxm-prod-device.streaming.siriusxm.com/"
717	        if url.hasPrefix(imageServer),
718	           let data = Data(base64Encoded: String(url.dropFirst(imageServer.count))),
719	           var spec = (try? JSONSerialization.jsonObject(with: data)) as? [String: Any] {
720	            spec["edits"] = [["resize": ["width": pixels, "height": pixels]]]
721	            guard let json = try? JSONSerialization.data(withJSONObject: spec, options: [.sortedKeys, .withoutEscapingSlashes])
722	            else { return url }
723	            return imageServer + json.base64EncodedString()
724	        }
725	        // Live album art comes as _m (450px); _t is 75px and _s 170px.
726	        if url.hasPrefix("http://albumart.siriusxm.com/"), url.hasSuffix("_m.jpg") {
727	            let size = pixels <= 75 ? "_t" : pixels <= 170 ? "_s" : "_m"
728	            return url.dropLast("_m.jpg".count) + size + ".jpg"
729	        }
730	        return url
731	    }
732	
733	    /// Decodes straight to a small bitmap, so neither the downloaded Data nor a full-size image is kept.
734	    nonisolated private static func thumbnail(_ data: Data, pixels: Int) -> CGImage? {
735	        guard let source = CGImageSourceCreateWithData(data as CFData, nil) else { return nil }
736	        let options: [CFString: Any] = [kCGImageSourceCreateThumbnailFromImageAlways: true,
737	                                        kCGImageSourceCreateThumbnailWithTransform: true,
738	                                        kCGImageSourceShouldCacheImmediately: true,
739	                                        kCGImageSourceThumbnailMaxPixelSize: pixels]
740	        return CGImageSourceCreateThumbnailAtIndex(source, 0, options as CFDictionary)
741	    }
742	}

744	/// Shows the image at `url` from `ArtCache`, loading it lazily at 2x `size` points. `content` gets
745	/// the image and whether it is album art (`isAlbumArt` belongs to `url`); nil when nothing is loaded.
746	private struct CachedArt<Content: View>: View {
747	    let url: String
748	    let size: CGFloat
749	    var isAlbumArt = false
750	    @ViewBuilder let content: (NSImage?, Bool) -> Content
751	    @State private var loaded: (url: String, image: NSImage?, isAlbumArt: Bool)?
752	
753	    var body: some View {
754	        let pixels = ArtCache.pixels(size)
755	        let image = loaded?.url == url ? loaded?.image : ArtCache.shared.cached(url, pixels: pixels)
756	        // Until the new art arrives, keep showing the previous art as it was (album art stays album
757	        // art): no flash of the logo or the StarPlayrX icon in between, in the header or in a row.
758	        Group {
759	            if let image { content(image, isAlbumArt) } else { content(loaded?.image, loaded?.isAlbumArt ?? false) }
760	        }
761	        .task(id: url) {
762	            let image = await ArtCache.shared.image(url, pixels: pixels)
763	            if !Task.isCancelled { loaded = (url, image, isAlbumArt) }
764	        }
765	    }
766	}

813	/// One line of text that never truncates: if it doesn't fit, it scrolls smoothly left in a loop
814	/// (pausing at the start of each pass); if it fits, it just sits still.
815	private struct MarqueeText: View {
816	    let text: String
817	    let font: Font
818	    var secondary = false
819	    @State private var textWidth: CGFloat = 0
820	
821	    private let gap: CGFloat = 40      // space between the end of the text and its repeat
822	    private let speed: CGFloat = 30    // points per second
823	    private let pause: Double = 2      // seconds held at the start of each pass
824	
825	    var body: some View {
826	        // Invisible line sets the height; the width comes from the parent.
827	        Text(" ").font(font).hidden()
828	            .frame(maxWidth: .infinity, alignment: .leading)
829	            .background(alignment: .leading) {
830	                Text(text).font(font).fixedSize().hidden()
831	                    .background(GeometryReader { g in
832	                        Color.clear.preference(key: MarqueeWidthKey.self, value: g.size.width)
833	                    })
834	            }
835	            .onPreferenceChange(MarqueeWidthKey.self) { textWidth = $0 }
836	            .overlay(alignment: .leading) {
837	                GeometryReader { geo in
838	                    if textWidth <= geo.size.width {
839	                        Text(text).font(font).fixedSize().foregroundStyle(secondary ? .secondary : .primary)
840	                    } else {
841	                        // Drawn once, then moved by Core Animation in the render server: no per-frame work
842	                        // in the app. (A TimelineView rebuilt every scrolling line 60 times a second.)
843	                        ScrollingStrip(content: AnyView(
844	                            HStack(spacing: gap) {
845	                                Text(text).font(font).fixedSize()
846	                                Text(text).font(font).fixedSize()
847	                            }
848	                            .foregroundStyle(secondary ? .secondary : .primary)),
849	                            key: "\(text)|\(secondary)", width: 2 * textWidth + gap, cycle: textWidth + gap, speed: speed, pause: pause)
850	                    }
851	                }
852	                .clipped()
853	            }
854	    }
855	}

857	/// Hosts `content` once and slides it left by `cycle` points on a repeating Core Animation, holding
858	/// at the start for `pause` seconds each pass. The content must repeat after `cycle` points.
859	private struct ScrollingStrip: NSViewRepresentable {
860	    let content: AnyView
861	    let key: String
862	    let cycle: CGFloat, speed: CGFloat, pause: Double
863	
864	    func makeNSView(context: Context) -> Strip { Strip() }
865	
866	    func updateNSView(_ view: Strip, context: Context) {
867	        view.configure(content: content, key: key, cycle: cycle, speed: speed, pause: pause)
868	    }
869	
870	    final class Strip: NSView {
871	        private let host = NSHostingView(rootView: AnyView(EmptyView()))
872	        private var key = ""
873	        private var width: CGFloat = 0, cycle: CGFloat = 0, speed: CGFloat = 30, pause: Double = 2
874	
875	        init() {
876	            super.init(frame: .zero)
877	            wantsLayer = true
878	            layer?.masksToBounds = true
879	            host.sizingOptions = []
880	            host.wantsLayer = true
881	            addSubview(host)
882	        }
883	        required init?(coder: NSCoder) { nil }
884	
885	        override func hitTest(_ point: NSPoint) -> NSView? { nil }   // clicks go to the row
886	
887	        func configure(content: AnyView, key: String, width: CGFloat, cycle: CGFloat, speed: CGFloat, pause: Double) {
888	            guard key != self.key || cycle != self.cycle || width != self.width else { return }
889	            self.key = key
890	            (self.width, self.cycle, self.speed, self.pause) = (width, cycle, speed, pause)
891	            host.rootView = content
892	            needsLayout = true
893	        }
894	
895	        override func layout() {
896	            super.layout()
897	            // Exactly the content's width (measured in SwiftUI; fittingSize isn't reliable here): any
898	            // wider and SwiftUI centers the two copies in it, so the text paused in from the left edge.
899	            let height = host.fittingSize.height > 0 ? host.fittingSize.height : bounds.height
900	            host.frame = NSRect(x: 0, y: (bounds.height - height) / 2, width: width, height: height)
901	            startAnimation()
902	        }
903	
904	        override func viewDidMoveToWindow() {
905	            super.viewDidMoveToWindow()
906	            if window != nil { startAnimation() }
907	        }
908	
909	        private func startAnimation() {
910	            guard let layer = host.layer, cycle > 0 else { return }
911	            let move = Double(cycle / speed)
912	            let period = move + pause
913	            let anim = CAKeyframeAnimation(keyPath: "transform.translation.x")
914	            anim.values = [0, 0, -cycle]
915	            anim.keyTimes = [0, NSNumber(value: pause / period), 1]
916	            anim.duration = period
917	            anim.repeatCount = .infinity
918	            anim.isRemovedOnCompletion = false
919	            layer.removeAnimation(forKey: "marquee")
920	            layer.add(anim, forKey: "marquee")
921	        }
922	    }
923	}
