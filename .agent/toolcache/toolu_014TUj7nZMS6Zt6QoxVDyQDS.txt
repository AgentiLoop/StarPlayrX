7 	import Foundation
8 	
9 	//MARK: Data Sync
10	internal func dataSync(endpoint: String, method: String, DataHandler: @escaping DataHandler ) {
11	 
12	    guard let url = URL(string: endpoint) else { DataHandler(.none); return}
13	    
14	    let semaphore = DispatchSemaphore(value: 0)
15	
16	    var urlReq = URLRequest(url: url)
17	    urlReq.httpMethod = "GET"
18	    urlReq.setValue(userAgent, forHTTPHeaderField: "User-Agent")
19	    urlReq.addValue("application/json", forHTTPHeaderField: "Content-Type")
20	    urlReq.timeoutInterval = TimeInterval(60)
21	    urlReq.cachePolicy = .reloadIgnoringLocalAndRemoteCacheData
22	    
23	    let task = URLSession.shared.dataTask(with: urlReq ) { ( data, _, _ ) in
24	        
25	        guard
26	            let d = data
27	        else { DataHandler(.none); return
28	        }
29	        
30	        DataHandler(d)
31	        
32	        semaphore.signal()
33	    }
34	    
35	    task.resume()
36	    _ = semaphore.wait(timeout: .distantFuture)
37	}
