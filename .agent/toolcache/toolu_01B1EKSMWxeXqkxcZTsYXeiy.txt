60 	    }
61 	
62 	    private func update(_ new: [Float]) {
63 	        guard tap != nil else { return }
64 	        let previous = levels
65 	        // Rise instantly, fall slowly.
66 	        levels = zip(levels, new).map { old, new in max(new, old * 0.8) }
67 	        // ~30 updates a second: quiet ≈ one colour cycle per 30 s, loud ≈ one per 3 s.
68 	        let energy = Double(new.reduce(0, +) / Float(new.count))
69 	        phase = (phase + 0.001 + energy * 0.02).truncatingRemainder(dividingBy: 1)
70 	        let bass = new.prefix(4).reduce(0, +) / 4
71 	        let kick = bass - bassAverage
72 	        bassAverage = bassAverage * 0.95 + bass * 0.05
73 	        pulse = max(Double(max(0, kick) * 4).clamped, pulse * 0.85)
74 	
75 	        // Bar colours blend the old rolling wave (global phase plus a slow wave across the bars)
76 	        // halfway toward a random palette. A strong kick (at most every ~3 s) or a rare random
77 	        // moment throws a new palette; each bar glides there at a pace set by its own band,
78 	        // onsets nudge it, and each bar is softened toward its neighbours so colours flow sideways.
79 	        paletteCooldown -= 1
80 	        if (kick > 0.25 && paletteCooldown <= 0) || Int.random(in: 0..<450) == 0 {
81 	            newPalette()
82 	            paletteCooldown = 90
83 	        }
84 	        let t = phase * 2 * .pi
85 	        let glided = hues.indices.map { i in
86 	            let x = Double(i) / Double(Self.bandCount)
87 	            let wave = phase + x * 0.3 + 0.08 * sin(x * 7 + t * 3)
88 	            let target = wave + 0.5 * Self.shortest(hueTargets[i] - wave)
89 	            let level = Double(new[i])
90 	            let onset = Double(max(0, new[i] - previous[i]))
91 	            return hues[i] + Self.shortest(target - hues[i]) * (0.01 + 0.08 * level) + onset * 0.04
92 	        }
93 	        hues = glided.indices.map { i in
94 	            let left = glided[max(0, i - 1)], right = glided[min(glided.count - 1, i + 1)]
95 	            let h = glided[i] + 0.25 * (Self.shortest(left - glided[i]) + Self.shortest(right - glided[i]))
96 	            return (h.truncatingRemainder(dividingBy: 1) + 1).truncatingRemainder(dividingBy: 1)
97 	        }
98 	
99 	        // Spectral centroid, 0 (bass-heavy) ... 1 (treble-heavy).
100	        let total = new.reduce(0, +)
101	        let weighted = new.enumerated().reduce(Float(0)) { $0 + Float($1.offset) * $1.element }
102	        let centroid = total > 0.01 ? Double(weighted / total) / Double(new.count - 1) : 0.5
103	        // New random spot on a strong kick, or now and then on its own.
104	        if kick > 0.15 || Int.random(in: 0..<90) == 0 { wanderTarget = .random(in: -1...1) }
105	        wander += (wanderTarget - wander) * 0.04
106	        let target = Swift.min(1, Swift.max(-1, (centroid * 2 - 1) * 1.5 + wander * 0.6))
107	        blobX += (target - blobX) * 0.08
108	    }
109	
110	    /// Hue difference taken the shortest way round the colour wheel, -0.5...0.5.
111	    private static func shortest(_ d: Double) -> Double {
112	        var d = d.truncatingRemainder(dividingBy: 1)
113	        if d > 0.5 { d -= 1 } else if d < -0.5 { d += 1 }
114	        return d
115	    }
116	
117	    /// Random palette in a random scheme, so the row never settles into one predictable cycle.
118	    private func newPalette() {
119	        let base = Double.random(in: 0..<1)
120	        let scheme = Int.random(in: 0..<5)
121	        let n = Self.bandCount
122	        hueTargets = (0..<n).map { i in
123	            let jitter = Double.random(in: -0.04...0.04)
124	            switch scheme {
125	            case 0: return base + jitter                                          // analogous
126	            case 1: return base + (Bool.random() ? 0 : 0.5) + jitter              // complementary
127	            case 2: return base + Double(Int.random(in: 0..<3)) / 3 + jitter      // triad
128	            case 3: return base + Double(i) / Double(n) * .random(in: 0.3...1)    // sweep
129	            default: return .random(in: 0..<1)                                    // confetti
130	            }
131	        }
132	    }
133	}

139	/// Picks bar colours that always stand out from the panel: WCAG relative luminance and
140	/// contrast ratio, lightening (dark mode) or darkening (light mode) a hue only as far as
141	/// needed to reach `minRatio`, so saturated colours survive wherever they already contrast.
142	enum ContrastColor {
143	    /// Approximate luminance of the menu/panel material behind the bars.
144	    static func background(_ scheme: ColorScheme) -> Double { scheme == .dark ? 0.025 : 0.84 }
145	
146	    static func color(hue: Double, saturation: Double, against bg: Double, minRatio: Double = 3.5) -> Color {
147	        let dark = bg < 0.5
148	        // Dark background: full brightness, then wash toward white. Light: darken.
149	        for step in 0...20 {
150	            let k = Double(step) / 20
151	            let s = dark ? saturation * (1 - 0.7 * k) : saturation
152	            let b = dark ? 1 : 1 - 0.75 * k
153	            let (r, g, bl) = rgb(hue, s, b)
154	            let l = 0.2126 * linear(r) + 0.7152 * linear(g) + 0.0722 * linear(bl)
155	            let ratio = (max(l, bg) + 0.05) / (min(l, bg) + 0.05)
156	            if ratio >= minRatio || step == 20 { return Color(red: r, green: g, blue: bl) }
157	        }
158	        return .primary
159	    }
160	
161	    private static func linear(_ c: Double) -> Double {
162	        c <= 0.04045 ? c / 12.92 : pow((c + 0.055) / 1.055, 2.4)
163	    }
164	
165	    private static func rgb(_ h: Double, _ s: Double, _ v: Double) -> (Double, Double, Double) {
166	        let h6 = (h - floor(h)) * 6
167	        let f = h6 - floor(h6)
168	        let p = v * (1 - s), q = v * (1 - s * f), t = v * (1 - s * (1 - f))
169	        switch Int(h6) % 6 {
170	        case 0: return (v, t, p)
171	        case 1: return (q, v, p)
172	        case 2: return (p, v, t)
173	        case 3: return (p, q, v)
174	        case 4: return (t, p, v)
175	        default: return (v, p, q)
176	        }
177	    }
178	}

180	/// Spectrum bars whose colours follow the music: each bar drifts at the pace of its own band,
181	/// onsets push it along, and bass kicks throw a fresh random palette. Every colour is checked
182	/// for contrast against the panel. Kicks also make the bars glow and a soft blob swell behind.
183	struct SpectrumView: View {
184	    @ObservedObject private var viz = Visualizer.shared
185	    @Environment(\.colorScheme) private var scheme
186	
187	    private func hue(_ x: Double) -> Double { (x.truncatingRemainder(dividingBy: 1) + 1).truncatingRemainder(dividingBy: 1) }
188	
189	    var body: some View {
190	        let bg = ContrastColor.background(scheme)
191	        HStack(alignment: .bottom, spacing: 2) {
192	            ForEach(viz.levels.indices, id: \.self) { i in
193	                let level = Double(viz.levels[i])
194	                let h = viz.hues[i]
195	                let base = ContrastColor.color(hue: h, saturation: 0.9, against: bg)
196	                let top = ContrastColor.color(hue: hue(h + 0.06 + 0.1 * level), saturation: 0.75, against: bg, minRatio: 4.5)
197	                RoundedRectangle(cornerRadius: 1.5)
198	                    .fill(LinearGradient(colors: [base, top], startPoint: .bottom, endPoint: .top))
199	                    .frame(height: max(2, CGFloat(level) * 28))
200	                    .shadow(color: top.opacity(0.4 + 0.6 * viz.pulse), radius: 1 + 4 * viz.pulse)
201	            }
202	        }
203	        .frame(height: 28, alignment: .bottom)
204	        .background {
205	            // The "lava": a blurred blob of the colour opposite the middle bar's, swelling on the beat.
206	            Ellipse()
207	                .fill(Color(hue: hue(viz.hues[Visualizer.bandCount / 2] + 0.5), saturation: 0.8, brightness: 1))
208	                .scaleEffect(x: 0.6 + 0.3 * sin(viz.phase * 2 * .pi * 2), y: 0.6 + 0.5 * viz.pulse)
209	                .offset(x: 80 * viz.blobX)
210	                .blur(radius: 14)
211	                .opacity(0.075 + 0.175 * viz.pulse)
212	        }
213	        .animation(.linear(duration: 0.05), value: viz.levels)
214	        .onAppear { viz.start() }
215	        .onDisappear { viz.stop() }
216	    }

220	/// The aggregate device shares its clock with the real output, so its IOProc must never miss a
221	/// cycle: a late IOProc overloads the output device too and the app's own playback skips.
222	/// The IOProc therefore runs on the HAL's real-time thread and only copies samples into a
223	/// preallocated ring buffer; the FFT runs ~30 times a second on a separate timer queue.
224	@available(macOS 14.2, *)
225	private final class ProcessTap {
226	    private var tapID = AudioObjectID(kAudioObjectUnknown)
227	    private var deviceID = AudioObjectID(kAudioObjectUnknown)
228	    private var procID: AudioDeviceIOProcID?
229	    private let queue = DispatchQueue(label: "StarPlayrX.tap", qos: .utility)
230	    private var timer: DispatchSourceTimer?
231	
232	    private let n = 1024
233	    private let fft = vDSP_create_fftsetup(10, FFTRadix(kFFTRadix2))!
234	    private var window: [Float]
235	    private var block: [Float]
236	    private let ringSize = 4096
237	    private let ring: UnsafeMutablePointer<Float>
238	    private var written = 0        // total samples written; guarded by lock
239	    private var lastRead = 0
240	    private let lock: UnsafeMutablePointer<os_unfair_lock>
241	    private let onLevels: ([Float]) -> Void
242	
243	    init?(onLevels: @escaping ([Float]) -> Void) {
244	        self.onLevels = onLevels
245	        window = [Float](repeating: 0, count: n)
246	        block = [Float](repeating: 0, count: n)
247	        vDSP_hann_window(&window, vDSP_Length(n), Int32(vDSP_HANN_NORM))
248	        ring = .allocate(capacity: ringSize)
249	        ring.initialize(repeating: 0, count: ringSize)
250	        lock = .allocate(capacity: 1)
251	        lock.initialize(to: os_unfair_lock())
252	
253	        guard let process = Self.ownProcessObject(), let outputUID = Self.defaultOutputUID() else { return nil }
254	
255	        let desc = CATapDescription(stereoMixdownOfProcesses: [process])
256	        desc.uuid = UUID()
257	        desc.isPrivate = true
258	        desc.muteBehavior = .unmuted
259	        guard AudioHardwareCreateProcessTap(desc, &tapID) == noErr else { return nil }
260	
261	        let aggregate: [String: Any] = [
262	            kAudioAggregateDeviceNameKey: "StarPlayrX Visualizer",
263	            kAudioAggregateDeviceUIDKey: UUID().uuidString,
264	            kAudioAggregateDeviceMainSubDeviceKey: outputUID,
265	            kAudioAggregateDeviceIsPrivateKey: true,
266	            kAudioAggregateDeviceIsStackedKey: false,
267	            kAudioAggregateDeviceTapAutoStartKey: true,
268	            kAudioAggregateDeviceSubDeviceListKey: [[kAudioSubDeviceUIDKey: outputUID]],
269	            kAudioAggregateDeviceTapListKey: [[kAudioSubTapDriftCompensationKey: true,
270	                                               kAudioSubTapUIDKey: desc.uuid.uuidString]]
271	        ]
272	        guard AudioHardwareCreateAggregateDevice(aggregate as CFDictionary, &deviceID) == noErr else {
273	            teardown()
274	            return nil
275	        }
276	        // nil queue: the block runs directly on the HAL's real-time IO thread.
277	        let status = AudioDeviceCreateIOProcIDWithBlock(&procID, deviceID, nil) { [unowned self] _, input, _, _, _ in
278	            self.receive(input)
279	        }
280	        guard status == noErr, AudioDeviceStart(deviceID, procID) == noErr else {
281	            teardown()
282	            return nil
283	        }
284	        let timer = DispatchSource.makeTimerSource(queue: queue)
285	        timer.schedule(deadline: .now(), repeating: .milliseconds(33))
286	        timer.setEventHandler { [weak self] in self?.analyze() }
287	        timer.resume()
288	        self.timer = timer
289	    }
290	
291	    deinit {
292	        teardown()
293	        vDSP_destroy_fftsetup(fft)
294	        ring.deallocate()
295	        lock.deallocate()
296	    }
297	
298	    private func teardown() {
299	        timer?.cancel()
300	        timer = nil
301	        if deviceID != kAudioObjectUnknown {
302	            AudioDeviceStop(deviceID, procID)
303	            if let procID { AudioDeviceDestroyIOProcID(deviceID, procID) }
304	            AudioHardwareDestroyAggregateDevice(deviceID)
305	            deviceID = AudioObjectID(kAudioObjectUnknown)
306	        }
307	        if tapID != kAudioObjectUnknown {
308	            AudioHardwareDestroyProcessTap(tapID)
309	            tapID = AudioObjectID(kAudioObjectUnknown)
310	        }
311	    }
312	
313	    /// Real-time thread: no allocation, no blocking. Mixes to mono into the ring buffer;
314	    /// if the analyzer holds the lock this cycle's samples are simply dropped.
315	    /// The aggregate lists the output device's own input streams first (BlackHole has one,
316	    /// silent without mic access) and the tap's stream last, so read only the last buffer.
317	    private func receive(_ input: UnsafePointer<AudioBufferList>) {
318	        guard let buffer = UnsafeMutableAudioBufferListPointer(UnsafeMutablePointer(mutating: input)).last,
319	              let data = buffer.mData, os_unfair_lock_trylock(lock) else { return }
320	        let channels = max(1, Int(buffer.mNumberChannels))
321	        let frames = Int(buffer.mDataByteSize) / MemoryLayout<Float>.size / channels
322	        let samples = data.assumingMemoryBound(to: Float.self)
323	        for f in 0..<frames {
324	            var sum: Float = 0
325	            for c in 0..<channels { sum += samples[f * channels + c] }
326	            ring[(written + f) % ringSize] = sum / Float(channels)
327	        }
328	        written += frames
329	        os_unfair_lock_unlock(lock)
330	    }
331	
332	    /// Timer queue: FFT of the newest 1024 samples, only when new audio has arrived.
333	    private func analyze() {
334	        os_unfair_lock_lock(lock)
335	        let end = written
336	        if end != lastRead, end >= n {
337	            for i in 0..<n { block[i] = ring[(end - n + i) % ringSize] }
338	        }
339	        os_unfair_lock_unlock(lock)
340	        guard end != lastRead, end >= n else { return }
341	        lastRead = end
342	        onLevels(bands(block))
343	    }
344	
345	    private func bands(_ block: [Float]) -> [Float] {
346	        let half = n / 2
347	        var windowed = [Float](repeating: 0, count: n)
348	        vDSP_vmul(block, 1, window, 1, &windowed, 1, vDSP_Length(n))
349	        var real = [Float](repeating: 0, count: half)
350	        var imag = [Float](repeating: 0, count: half)
351	        var mags = [Float](repeating: 0, count: half)
352	        real.withUnsafeMutableBufferPointer { rp in
353	            imag.withUnsafeMutableBufferPointer { ip in
354	                var split = DSPSplitComplex(realp: rp.baseAddress!, imagp: ip.baseAddress!)
355	                windowed.withUnsafeBufferPointer { wp in
356	                    wp.baseAddress!.withMemoryRebound(to: DSPComplex.self, capacity: half) {
357	                        vDSP_ctoz($0, 2, &split, 1, vDSP_Length(half))
358	                    }
359	                }
360	                vDSP_fft_zrip(fft, &split, 1, 10, FFTDirection(FFT_FORWARD))
361	                vDSP_zvabs(&split, 1, &mags, 1, vDSP_Length(half))
362	            }
363	        }
364	        // Log-spaced bands from bin 2 (~90 Hz at 48 kHz) up to Nyquist.
365	        let count = Visualizer.bandCount
366	        return (0..<count).map { b in
367	            let lo = Int(2 * pow(Float(half / 2), Float(b) / Float(count)))
368	            let hi = max(lo + 1, min(half, Int(2 * pow(Float(half / 2), Float(b + 1) / Float(count)))))
369	            let peak = mags[lo..<hi].max() ?? 0
370	            let db = 20 * log10(peak / Float(n) + 1e-9)
371	            return max(0, min(1, (db + 50) / 50))
372	        }
373	    }

375	    private static func ownProcessObject() -> AudioObjectID? {
376	        var pid = getpid()
377	        var address = AudioObjectPropertyAddress(mSelector: kAudioHardwarePropertyTranslatePIDToProcessObject,
378	                                                 mScope: kAudioObjectPropertyScopeGlobal,
379	                                                 mElement: kAudioObjectPropertyElementMain)
380	        var object = AudioObjectID(kAudioObjectUnknown)
381	        var size = UInt32(MemoryLayout<AudioObjectID>.size)
382	        let status = AudioObjectGetPropertyData(AudioObjectID(kAudioObjectSystemObject), &address,
383	                                                UInt32(MemoryLayout<pid_t>.size), &pid, &size, &object)
384	        return status == noErr && object != kAudioObjectUnknown ? object : nil
385	    }
386	
387	    private static func defaultOutputUID() -> String? {
388	        var address = AudioObjectPropertyAddress(mSelector: kAudioHardwarePropertyDefaultOutputDevice,
389	                                                 mScope: kAudioObjectPropertyScopeGlobal,
390	                                                 mElement: kAudioObjectPropertyElementMain)
391	        var device = AudioObjectID(kAudioObjectUnknown)
392	        var size = UInt32(MemoryLayout<AudioObjectID>.size)
393	        guard AudioObjectGetPropertyData(AudioObjectID(kAudioObjectSystemObject), &address, 0, nil, &size, &device) == noErr
394	        else { return nil }
395	        address.mSelector = kAudioDevicePropertyDeviceUID
396	        var uid: Unmanaged<CFString>?
397	        size = UInt32(MemoryLayout<Unmanaged<CFString>?>.size)
398	        guard AudioObjectGetPropertyData(device, &address, 0, nil, &size, &uid) == noErr else { return nil }
399	        return uid?.takeRetainedValue() as String?
400	    }
