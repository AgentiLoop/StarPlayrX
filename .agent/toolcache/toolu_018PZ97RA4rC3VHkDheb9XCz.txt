65 	    private func startServerIfNeeded() async {
66 	        if serverStarted, await text("ping") == "pong" { return }
67 	        for p in port..<65000 where portIsOpen(UInt16(p)) {
68 	            port = UInt16(p)
69 	            break
70 	        }
71 	        startServer(port)
72 	        preflightConfig(location: localeIsCA ? "CA" : "US")
73 	        for _ in 0..<20 {
74 	            if await text("ping") == "pong" { break }
75 	            try? await Task.sleep(nanoseconds: 100_000_000)
76 	        }
77 	        serverStarted = true

82 	    func login(user: String, pass: String) async -> Bool {
83 	        isBusy = true
84 	        channelsLoaded = false
85 	        defer { isBusy = false }
86 	        status = "Starting…"
87 	        await startServerIfNeeded()
88 	        _ = await text(localeIsCA ? "ca" : "us")
89 	
90 	        status = "Signing in…"
91 	        let result = await post("login", ["user": user, "pass": pass])
92 	        // StarPlayrRadioKit caches the password in UserDefaults; the Mac app keeps it only in the Keychain.
93 	        UserDefaults.standard.removeObject(forKey: "pass")
94 	
95 	        guard let result, result["success"] as? Bool == true else {
96 	            status = (result?["message"] as? String) ?? "Login failed. Check your connection."
97 	            return false
98 	        }
99 	        Keychain.save(user: user, pass: pass)
100	
101	        status = "Loading channels…"
102	        _ = await post("session", ["channelid": "siriushits1"])
103	        guard let data = await post("channels", ["channeltype": ""])?["data"] as? [String: Any], !data.isEmpty else {
104	            status = "Error reading channels."
105	            return false
106	        }
107	
108	        channels = data.compactMap { key, value in
109	            guard let v = value as? [String: Any],
110	                  let number = v["channelNumber"] as? String,
111	                  var name = v["name"] as? String else { return nil }
112	            for prefix in ["SiriusXM ", "Sirius XM ", "Sirius ", "SXM ", "SPX "] {
113	                name = name.replacingOccurrences(of: prefix, with: "")
114	            }
115	            return Channel(id: number, name: name,
116	                           category: v["category"] as? String ?? "",
117	                           imageURL: v["mediumImage"] as? String ?? "")
118	        }
119	        .sorted { $0.id.localizedStandardCompare($1.id) == .orderedAscending }
120	
121	        isLoggedIn = true
122	        status = "Signed in"
123	        if let last = UserDefaults.standard.string(forKey: "lastChannel") {
124	            current = channels.first { $0.id == last }
125	            // Auto play the last channel played (Xtra channels auto play once they load below).
126	            if let current, !isPlaying { play(current) }
127	        }
128	        await updatePDT()
129	        startPDTTimer()
130	        Task { await loadXtraChannels(user: user, pass: pass) }
131	        return true
132	    }

134	    /// Xtra channels come from SiriusXM's newer API; if that fails the live channels still work.
135	    private func loadXtraChannels(user: String, pass: String) async {
136	        defer { if isLoggedIn { channelsLoaded = true } }
137	        do {
138	            try await xtra.signIn(user: user, pass: pass)
139	            let all = try await xtra.channels()
140	            let xtras = all.filter(\.isXtra).map {
141	                Channel(id: $0.number, name: $0.name, category: $0.genre, imageURL: $0.imageURL, xtraID: $0.id)
142	            }
143	            // Live channels get the same full square tile art as Xtra channels when the newer API has it.
144	            var tiles = [String: String]()
145	            for ch in all where !ch.isXtra && !ch.imageURL.isEmpty { tiles[ch.number] = ch.imageURL }
146	            guard isLoggedIn else { return }
147	            liveIDs = Dictionary(all.filter { !$0.isXtra }.map { ($0.number, $0.id) }) { first, _ in first }
148	            let withTiles = channels.map { ch in
149	                tiles[ch.id].map { Channel(id: ch.id, name: ch.name, category: ch.category, imageURL: $0, xtraID: ch.xtraID) } ?? ch
150	            }
151	            let live = Set(withTiles.map(\.id))
152	            channels = (withTiles + xtras.filter { !live.contains($0.id) })
153	                .sorted { $0.id.localizedStandardCompare($1.id) == .orderedAscending }
154	            let wasNil = current == nil
155	            if let id = current?.id ?? UserDefaults.standard.string(forKey: "lastChannel") {
156	                current = channels.first { $0.id == id }   // pick up the new art for the current channel too
157	                // Auto play a last-played Xtra channel, which wasn't available at login.
158	                if wasNil, let current, current.xtraID != nil, !isPlaying { play(current) }
159	            }
160	            await refreshXtraNowPlaying()
161	        } catch {
162	            print("Xtra channels unavailable: \(error.localizedDescription)")
163	        }
164	    }

186	    func play(_ channel: Channel) {
187	        haltXtra()
188	        current = channel
189	        UserDefaults.standard.set(channel.id, forKey: "lastChannel")
190	        if let xtraID = channel.xtraID {
191	            player.pause()
192	            player.replaceCurrentItem(with: nil)
193	            playXtra(channel, id: xtraID)
194	            return
195	        }
196	        Task {
197	            await startServerIfNeeded()
198	            guard let url = URL(string: "http://localhost:\(port)/api/v3/m3u/\(channel.id).m3u8") else { return }
199	            let item = AVPlayerItem(url: url)
200	            player.replaceCurrentItem(with: item)
201	            player.play()
202	            isPlaying = true
203	            // AVPlayer starts several segments behind live, so the audio would lag the on-air song
204	            // the channel list shows. Jump to the live edge so tuning in plays what the list says.
205	            var tries = 0
206	            while (item.status != .readyToPlay || item.seekableTimeRanges.isEmpty), tries < 100 {
207	                try? await Task.sleep(nanoseconds: 100_000_000)
208	                tries += 1
209	            }
210	            guard player.currentItem === item else { return }
211	            if let range = item.seekableTimeRanges.last?.timeRangeValue {
212	                let edge = CMTimeSubtract(CMTimeRangeGetEnd(range), CMTime(seconds: 2, preferredTimescale: 1000))
213	                await item.seek(to: max(edge, range.start))
214	            }
215	            print("LIVE lag \(item.currentDate().map { Int(Date().timeIntervalSince($0)) } ?? -1)s")
216	            await updatePDT()
217	        }
218	    }

234	    private func playXtra(_ channel: Channel, id: String) {
235	        isPlaying = true
236	        // Tuning again returns a different track than the peek, so start with the peeked track
237	        // the channel list is showing (from its beginning), then continue with the tuned queue.
238	        let peeked = xtraPeekedTrack.removeValue(forKey: channel.id).map {
239	            XtraTrack(id: $0.id, artist: $0.artist, title: $0.title, url: $0.url, keyID: $0.keyID,
240	                      imageURL: $0.imageURL, startOffset: 0)
241	        }
242	        print("XTRA play \(channel.id) shown=\(nowPlaying[channel.id]?.song ?? "nil") peeked=\(peeked?.title ?? "nil")")
243	        if peeked == nil {
244	            xtraNowPlaying[channel.id] = nil
245	            nowPlaying[channel.id] = nil
246	        }
247	        updateNowPlayingInfo()
248	        xtraTask = Task {
249	            var pending = [XtraTrack]()
250	            var token: String?
251	            var seen = Set<String>()
252	            var first = true
253	            if let peeked {
254	                seen.insert(peeked.id)
255	                do {
256	                    let file = try await xtra.download(peeked, into: xtraFolder)
257	                    if !Task.isCancelled {
258	                        let item = AVPlayerItem(url: file)
259	                        xtraTracks[item] = peeked
260	                        xtraPlayer.insert(item, after: nil)
261	                        first = false
262	                        xtraPlayer.play()
263	                        xtraItemChanged()
264	                    }
265	                } catch {
266	                    print("XTRA peeked download failed \(channel.id) \(peeked.title): \(error.localizedDescription) url=\(peeked.url)")
267	                }
268	            }
269	            do {
270	                while !Task.isCancelled {
271	                    // Keep one track playing and one ready behind it.
272	                    if xtraPlayer.items().count > 1 {
273	                        try await Task.sleep(nanoseconds: 2_000_000_000)
274	                        continue
275	                    }
276	                    if pending.isEmpty {
277	                        let batch = try await xtra.tune(channelID: id, sequenceToken: token)
278	                        token = batch.token
279	                        pending = batch.tracks.filter { seen.insert($0.id).inserted }
280	                        if pending.isEmpty { try await Task.sleep(nanoseconds: 5_000_000_000); continue }
281	                    }
282	                    let track = pending.removeFirst()
283	                    let file = try await xtra.download(track, into: xtraFolder)
284	                    guard !Task.isCancelled else { break }
285	                    let item = AVPlayerItem(url: file)
286	                    xtraTracks[item] = track
287	                    xtraPlayer.insert(item, after: nil)
288	                    if first {
289	                        first = false
290	                        if track.startOffset > 0 {
291	                            await xtraPlayer.seek(to: CMTime(seconds: track.startOffset, preferredTimescale: 1000))
292	                        }
293	                        xtraPlayer.play()
294	                        xtraItemChanged()
295	                    }
296	                }
297	            } catch is CancellationError {
298	            } catch {
299	                print("Xtra playback error: \(error.localizedDescription)")
300	                if first, current?.id == channel.id { stop() }
301	            }
302	        }
303	    }

305	    private func haltXtra() {
306	        // The list keeps showing the song that was playing, so tuning back in should play that song.
307	        if let item = xtraPlayer.currentItem, let track = xtraTracks[item], let current, current.xtraID != nil {
308	            xtraPeekedTrack[current.id] = track
309	        }
310	        xtraTask?.cancel()
311	        xtraTask = nil
312	        xtraPlayer.pause()
313	        xtraPlayer.removeAllItems()
314	        xtraTracks = [:]
315	        try? FileManager.default.removeItem(at: xtraFolder)
316	    }

318	    /// Shows the queued track that just started and deletes files of tracks that have finished.
319	    fileprivate func xtraItemChanged() {
320	        let queued = Set(xtraPlayer.items())
321	        for (item, track) in xtraTracks where !queued.contains(item) {
322	            xtraTracks[item] = nil
323	            try? FileManager.default.removeItem(at: xtraFolder.appendingPathComponent("\(track.id).aac"))
324	        }
325	        guard let item = xtraPlayer.currentItem, let track = xtraTracks[item], let current, current.xtraID != nil else { return }
326	        print("XTRA playing \(current.id) \(track.artist) – \(track.title)")
327	        let np = NowPlaying(artist: track.artist, song: track.title, image: track.imageURL)
328	        xtraNowPlaying[current.id] = np
329	        nowPlaying[current.id] = np
330	        updateNowPlayingInfo()
331	    }

342	    func updatePDT() async {
343	        guard let cache = await get("pdt")?["data"] as? [String: Any] else { return }
344	        var np = [String: NowPlaying]()
345	        for (key, value) in cache {
346	            if let v = value as? [String: String] {
347	                np[key] = Self.keepingArt(NowPlaying(artist: v["artist"] ?? "", song: v["song"] ?? "", image: v["image"] ?? ""),
348	                                          of: nowPlaying[key])
349	            }
350	        }
351	        // The lookAround feed has every live channel's current cut, mostly with album art, in one call;
352	        // it wins over PDT, which only has art for the tuned channel. PDT stays as the fallback.
353	        if !liveIDs.isEmpty, let live = try? await xtra.liveNowPlaying() {
354	            for (number, id) in liveIDs {
355	                guard let la = live[id] else { continue }
356	                np[number] = Self.keepingArt(Self.keepingArt(la, of: np[number]), of: nowPlaying[number])
357	            }
358	        }
359	        // lookAround says what is on air now, but the player runs behind live. For the tuned live channel,
360	        // show the cut at the player's position in the stream instead, from the channel's own timeline.
361	        if isPlaying, let current, current.xtraID == nil, let id = liveIDs[current.id],
362	           let at = player.currentItem?.currentDate() {
363	            // Refetch on a new channel, or once playback reaches the newest known cut (its end isn't known yet).
364	            if liveTimeline.channel != current.id
365	                || (!liveTimeline.cuts.contains { $0.start > at } && Date().timeIntervalSince(liveTimeline.fetched) > 30),
366	               let cuts = try? await xtra.liveTimeline(channelID: id) {
367	                liveTimeline = (current.id, Date(), cuts)
368	            }
369	            if liveTimeline.channel == current.id, let cut = liveTimeline.cuts.last(where: { $0.start <= at }) {
370	                np[current.id] = Self.keepingArt(cut.nowPlaying, of: nowPlaying[current.id])
371	            }
372	        }
373	        nowPlaying = np.merging(xtraNowPlaying) { live, _ in live }
374	        updateNowPlayingInfo()
375	        Task { await refreshXtraNowPlaying() }
376	    }
377	    private var liveTimeline = (channel: "", fetched: Date.distantPast, cuts: [(start: Date, nowPlaying: NowPlaying)]())
378	
379	    /// Xtra channels have no PDT feed, so ask each idle one what it would play if tuned now.
380	    /// Channels never peeked are all filled at once; after that the stalest are refreshed
381	    /// a batch at a time on every PDT tick. Each result is shown as soon as it arrives.
382	    private func refreshXtraNowPlaying() async {
383	        guard isLoggedIn, !xtraPeeking else { return }
384	        xtraPeeking = true
385	        defer { xtraPeeking = false }
386	        let stale = Date().addingTimeInterval(-180)
387	        let idle = channels.filter { $0.xtraID != nil && !isPlayingXtra($0.id) }
388	        let unpeeked = idle.filter { xtraPeekedAt[$0.id] == nil }
389	        let batch = !unpeeked.isEmpty ? unpeeked : Array(idle
390	            .filter { xtraPeekedAt[$0.id]! < stale }
391	            .sorted { xtraPeekedAt[$0.id]! < xtraPeekedAt[$1.id]! }
392	            .prefix(30))
393	        guard !batch.isEmpty else { return }
394	        let client = xtra
395	        await withTaskGroup(of: (String, XtraTrack?).self) { group in
396	            for (i, ch) in batch.enumerated() {
397	                if i >= 12, let r = await group.next() { applyXtraPeek(r.0, r.1) }
398	                group.addTask { (ch.id, try? await client.peek(channelID: ch.xtraID!)) }
399	            }
400	            for await r in group { applyXtraPeek(r.0, r.1) }
401	        }
402	    }

404	    /// Peek results arrive many per second; publishing each one redraws the whole channel
405	    /// list and stalls scrolling, so they're batched and published at most every 2 seconds.
406	    private func applyXtraPeek(_ id: String, _ track: XtraTrack?) {
407	        guard isLoggedIn else { return }
408	        xtraPeekedAt[id] = Date()
409	        guard let track, !isPlayingXtra(id) else { return }
410	        // The played track, xtraNowPlaying and the list text are committed together on flush,
411	        // so tapping a channel always plays the song its row is showing.
412	        pendingPeeks[id] = track
413	        guard !peekFlushScheduled else { return }
414	        peekFlushScheduled = true
415	        Task {
416	            try? await Task.sleep(for: .seconds(2))
417	            peekFlushScheduled = false
418	            let tracks = pendingPeeks.filter { !isPlayingXtra($0.key) }
419	            pendingPeeks.removeAll()
420	            guard isLoggedIn, !tracks.isEmpty else { return }
421	            let batch = Dictionary(uniqueKeysWithValues: tracks.map { id, track in
422	                (id, Self.keepingArt(NowPlaying(artist: track.artist, song: track.title, image: track.imageURL), of: nowPlaying[id]))
423	            })
424	            xtraPeekedTrack.merge(tracks) { _, new in new }
425	            xtraNowPlaying.merge(batch) { _, new in new }
426	            nowPlaying.merge(batch) { _, new in new }
427	        }
428	    }
429	    /// The PDT feed only has album art for songs the server has just seen on a now-playing-live
430	    /// fetch (it forgets them on the next one), so a later poll of the same song often comes back
431	    /// without art. Keep the album art until the song changes, instead of flipping back to the logo.
432	    nonisolated static func keepingArt(_ new: NowPlaying, of old: NowPlaying?) -> NowPlaying {
433	        guard new.image.isEmpty, let old, !old.image.isEmpty, old.artist == new.artist, old.song == new.song else { return new }
434	        var kept = new
435	        kept.image = old.image
436	        return kept
437	    }

462	    private func updateNowPlayingInfo() {
463	        let center = MPNowPlayingInfoCenter.default()
464	        guard let current else { center.nowPlayingInfo = nil; return }
465	        let np = nowPlaying[current.id]
466	        var info: [String: Any] = [
467	            MPMediaItemPropertyTitle: np?.song ?? current.name,
468	            MPMediaItemPropertyArtist: np?.artist ?? "",
469	            MPMediaItemPropertyAlbumTitle: current.name,
470	            MPNowPlayingInfoPropertyIsLiveStream: true
471	        ]
472	        // Same art as the menu's Now Playing tile: album art when there is some, else the channel logo.
473	        let url = np?.image.isEmpty == false ? np!.image : current.imageURL
474	        if url == artworkURL, let artwork { info[MPMediaItemPropertyArtwork] = artwork }
475	        center.nowPlayingInfo = info
476	        center.playbackState = isPlaying ? .playing : .paused
477	        guard url != artworkURL else { return }
478	        artworkURL = url
479	        artwork = nil
480	        Task {
481	            guard let image = await ArtCache.shared.image(url, pixels: ArtCache.pixels(150)),
482	                  artworkURL == url else { return }
483	            artwork = MPMediaItemArtwork(boundsSize: image.size) { _ in image }
484	            updateNowPlayingInfo()
485	        }
486	    }

506	    private func text(_ path: String) async -> String? {
507	        guard let url = URL(string: "\(base)/\(path)") else { return nil }
508	        var req = URLRequest(url: url, cachePolicy: .reloadIgnoringLocalAndRemoteCacheData, timeoutInterval: 2)
509	        req.httpMethod = "GET"
510	        guard let (data, _) = try? await URLSession.shared.data(for: req) else { return nil }
511	        return String(data: data, encoding: .utf8)
512	    }

514	    private func get(_ path: String) async -> [String: Any]? {
515	        guard let url = URL(string: "\(base)/\(path)") else { return nil }
516	        let req = URLRequest(url: url, cachePolicy: .reloadIgnoringLocalAndRemoteCacheData, timeoutInterval: 60)
517	        guard let (data, _) = try? await URLSession.shared.data(for: req) else { return nil }
518	        return (try? JSONSerialization.jsonObject(with: data)) as? [String: Any]
519	    }

521	    private func post(_ path: String, _ body: [String: Any]) async -> [String: Any]? {
522	        guard let url = URL(string: "\(base)/\(path)") else { return nil }
523	        var req = URLRequest(url: url, cachePolicy: .reloadIgnoringLocalAndRemoteCacheData, timeoutInterval: 60)
524	        req.httpMethod = "POST"
525	        req.httpBody = try? JSONSerialization.data(withJSONObject: body)
526	        req.addValue("application/json", forHTTPHeaderField: "Content-Type")
527	        guard let (data, _) = try? await URLSession.shared.data(for: req) else { return nil }
528	        return (try? JSONSerialization.jsonObject(with: data)) as? [String: Any]
529	    }

533	    private func portIsOpen(_ port: in_port_t) -> Bool {
534	        let fd = socket(AF_INET, SOCK_STREAM, 0)
535	        guard fd != -1 else { return false }
536	        defer { Darwin.shutdown(fd, SHUT_RDWR); close(fd) }
537	
538	        var addr = sockaddr_in()
539	        addr.sin_len = __uint8_t(MemoryLayout<sockaddr_in>.size)
540	        addr.sin_family = sa_family_t(AF_INET)
541	        addr.sin_port = port.bigEndian
542	        addr.sin_addr = in_addr(s_addr: inet_addr("127.0.0.1"))
543	
544	        let bound = withUnsafePointer(to: &addr) {
545	            $0.withMemoryRebound(to: sockaddr.self, capacity: 1) {
546	                Darwin.bind(fd, $0, socklen_t(MemoryLayout<sockaddr_in>.size))
547	            }
548	        }
549	        return bound != -1 && listen(fd, SOMAXCONN) != -1
550	    }
