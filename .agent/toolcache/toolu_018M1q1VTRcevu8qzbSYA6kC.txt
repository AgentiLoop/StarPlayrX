134	    /// Decodes an AAC file and hands `emit` the bars of every 1024 samples, about a second at a time:
135	    /// (seconds per frame, index of the first frame, frames). `emit` returns false to stop.
136	    private static func decode(_ url: URL, from seconds: Double = 0,
137	                               emit: (Double, Int, [[Float]]) -> Bool) {
138	        guard let file = try? AVAudioFile(forReading: url) else { return }
139	        let format = file.processingFormat
140	        let rate = format.sampleRate
141	        let channels = Int(format.channelCount)
142	        var index = Int(seconds * rate) / n
143	        file.framePosition = AVAudioFramePosition(index * n)
144	        let chunk = AVAudioFrameCount(n * 43)
145	        guard rate > 0, channels > 0, file.framePosition < file.length,
146	              let buffer = AVAudioPCMBuffer(pcmFormat: format, frameCapacity: chunk) else { return }
147	        let bands = Bands()
148	        while (try? file.read(into: buffer, frameCount: chunk)) != nil, buffer.frameLength > 0,
149	              let data = buffer.floatChannelData {
150	            let length = Int(buffer.frameLength)
151	            var mono = Array(UnsafeBufferPointer(start: data[0], count: length))
152	            for c in 1..<channels { mono = vDSP.add(mono, UnsafeBufferPointer(start: data[c], count: length)) }
153	            if channels > 1 { mono = vDSP.multiply(1 / Float(channels), mono) }
154	            let frames = stride(from: 0, through: length - n, by: n).map { bands.levels(mono[$0..<$0 + n]) }
155	            guard emit(Double(n) / rate, index, frames) else { return }
156	            index += frames.count
157	        }
158	    }

160	    /// FFT of one 1024-sample block into `Visualizer.bandCount` bar heights (0...1).
161	    private final class Bands {
162	        private let fft = vDSP_create_fftsetup(10, FFTRadix(kFFTRadix2))!
163	        private var window = [Float](repeating: 0, count: SpectrumTimeline.n)
164	
165	        init() { vDSP_hann_window(&window, vDSP_Length(SpectrumTimeline.n), Int32(vDSP_HANN_NORM)) }
166	        deinit { vDSP_destroy_fftsetup(fft) }
167	
168	        func levels(_ block: ArraySlice<Float>) -> [Float] {
169	            let n = SpectrumTimeline.n
170	            let half = n / 2
171	            let windowed = vDSP.multiply(block, window)
172	            var real = [Float](repeating: 0, count: half)
173	            var imag = [Float](repeating: 0, count: half)
174	            var mags = [Float](repeating: 0, count: half)
175	            real.withUnsafeMutableBufferPointer { rp in
176	                imag.withUnsafeMutableBufferPointer { ip in
177	                    var split = DSPSplitComplex(realp: rp.baseAddress!, imagp: ip.baseAddress!)
178	                    windowed.withUnsafeBufferPointer { wp in
179	                        wp.baseAddress!.withMemoryRebound(to: DSPComplex.self, capacity: half) {
180	                            vDSP_ctoz($0, 2, &split, 1, vDSP_Length(half))
181	                        }
182	                    }
183	                    vDSP_fft_zrip(fft, &split, 1, 10, FFTDirection(FFT_FORWARD))
184	                    vDSP_zvabs(&split, 1, &mags, 1, vDSP_Length(half))
185	                }
186	            }
187	            // Log-spaced bands from bin 2 (~90 Hz) up to Nyquist.
188	            let count = Visualizer.bandCount
189	            return (0..<count).map { b in
190	                let lo = Int(2 * pow(Float(half / 2), Float(b) / Float(count)))
191	                let hi = max(lo + 1, min(half, Int(2 * pow(Float(half / 2), Float(b + 1) / Float(count)))))
192	                let peak = mags[lo..<hi].max() ?? 0
193	                let db = 20 * log10(peak / Float(n) + 1e-9)
194	                return max(0, min(1, (db + 50) / 50))
195	            }
196	        }
